#+title: Manifolding Atlas Links Plugin
#+priority: 15

Guards file and heading deletion when incoming or outgoing =id:= links exist.
Guards deletion of untracked files and missing links.
Provides restore command via Git history.
Uses native =manifolding-atlas-db-query-links-to= and =manifolding-atlas-db-query-links-from= API (v2.1+).

**** Added-entry jump to property in capture
RET on an Added Heading Blueprints entry opens the note in an
org-capture buffer positioned on the added :KEY: line (widened,
point on the value). The admin TODO is left untouched — mark DONE
manually. Deleted Heading Blueprints keeps plain file-open.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-links--added-property-at-point ()
  "Return (KEY . NOTE-FILE) from the Added entry heading at point, or nil.
Reads :BLUEPRINT_KEY: and :NOTE_FILE: from the current heading drawer."
  (when (org-at-heading-p)
    (let ((key (org-entry-get nil "BLUEPRINT_KEY"))
          (file (org-entry-get nil "NOTE_FILE")))
      (when (and key file (not (string-empty-p key)) (file-exists-p file))
        (cons key file)))))

(defun my/manifolding-atlas-links--capture-at-property (key file)
  "Open FILE in an org-capture buffer positioned after the :KEY: colon.
Widened buffer (`:unnarrowed t'), point lands on the value for editing.
The admin TODO is left alone — mark it DONE manually."
  (let ((org-capture-templates
         `(("p" "Edit added property"
            plain
            (file+function
             ,file
             (lambda ()
               (goto-char (point-min))
               (if (re-search-forward
                    (format "^[ \t]*:%s:[ \t]*" (regexp-quote ,key))
                    nil t)
                   (goto-char (match-end 0))
                 (goto-char (point-min)))))
            "%?"
            :kill-buffer t :unnarrowed t :empty-lines 0))))
    (org-capture nil "p")))

(defun my/manifolding-atlas-added--around-open (orig-fun &rest args)
  "Wrap org-open-at-point: from Added Heading Blueprints, RET opens the
note in an org-capture buffer on the added :KEY: line.
All other buffers fall through to ORIG-FUN untouched."
  (let ((added-path (ignore-errors
                      (expand-file-name my/manifolding-atlas-links--added-blueprints-file
                                       (my/manifolding-atlas-root-dir)))))
    (if (and buffer-file-name added-path
             (string= (expand-file-name buffer-file-name) added-path)
             (my/manifolding-atlas-links--added-property-at-point))
        (let ((kv (my/manifolding-atlas-links--added-property-at-point)))
          (delete-other-windows)
          (my/manifolding-atlas-links--capture-at-property (car kv) (cdr kv)))
      (apply orig-fun args))))
(advice-add 'org-open-at-point :around #'my/manifolding-atlas-added--around-open)

(cl-defun my/manifolding-atlas-links--write-deleted-entry (file &optional unfoldings)
  "Append a heading for deleted FILE to ReManifold.
UNFOLDINGS is a list of relative paths of unfolding notes deleted alongside FILE,
recorded in the :DELETED_UNFOLDINGS: unfolding for later restore."
  (let ((path (my/manifolding-atlas-links--ensure-deleted-files-file))
        root rel deleted-id deleted-title heading)
    (setq root (expand-file-name (or (vc-root-dir) "")
                                 (my/manifolding-atlas-root-dir)))
    (setq rel (file-relative-name (expand-file-name file) root))
    (when (string-prefix-p ".." rel) (cl-return-from my/manifolding-atlas-links--write-deleted-entry nil))
    (setq deleted-id (org-entry-get (point-min) "ID"))
    (setq deleted-title (cadr (assoc "TITLE"
                                     (org-collect-keywords (list "TITLE")))))
    (setq heading (or deleted-title (file-name-base file)))
    (with-current-buffer (find-file-noselect path)
      (goto-char (point-max))
      (insert (format "\n* TODO [[restore:][%s]]\n" heading))
      (insert (my/manifolding-atlas-links--deleted-entry-props rel deleted-id unfoldings))
      (save-buffer)
      (kill-buffer (current-buffer)))))

*** Unfolding Cleanup on Delete

**** Unfolding id detection
#+begin_src emacs-lisp
(defun my/manifolding-atlas-links--unfolding-id-in-file-p (id file)
  "Non-nil when FILE contains an [[id:ID][...]] link."
  (let ((re (format "\\[\\[id:%s\\]\\[" (regexp-quote id))))
    (with-temp-buffer
      (insert-file-contents file)
      (goto-char (point-min))
      (re-search-forward re nil t))))

(defun my/manifolding-atlas-links--unfolding-id-p (id)
  "Non-nil when ID names an unfolding note under admin/unfoldings/."
  (when-let ((note (manifolding-atlas-db-get-by-id id)))
    (let ((path (manifolding-atlas-note-path note)))
      (and path
           (file-in-directory-p path
                                (expand-file-name "admin/unfoldings/"
                                                  (my/manifolding-atlas-root-dir)))))))

(defun my/manifolding-atlas-links--note-unfolding-ids (file)
  "Return list of unfolding note IDs referenced by FILE.
Scans FILE for [[id:ID][...]] links that resolve to a note under
admin/unfoldings/. Returns the list of matching IDs."
  (let ((root (my/manifolding-atlas-root-dir))
        unfolding-ids)
    (when (file-exists-p file)
      (with-temp-buffer
        (insert-file-contents file)
        (goto-char (point-min))
        (while (re-search-forward "\\[\\[id:\\([^]]+\\)\\]\\[" nil t)
          (let ((id (match-string-no-properties 1)))
            (when (my/manifolding-atlas-links--unfolding-id-p id)
              (push id unfolding-ids))))))
    (delete-dups unfolding-ids)))
#+end_src

**** Orphan detection
Skips DELETED-FILE itself and files under admin/unfoldings/.
#+begin_src emacs-lisp
 (defun my/manifolding-atlas-links--unfolding-orphaned-p (unfolding-id deleted-file)
  "Non-nil when UNFOLDING-ID is referenced only by DELETED-FILE.
For scoped unfoldings (with UNFOLDING_PARENT), checks parent ID directly;
for legacy shared unfoldings (no UNFOLDING_PARENT), falls back to full vault scan."
  (let* ((unfolding-note (ignore-errors (manifolding-atlas-db-get-by-id unfolding-id)))
         (unfolding-parent (and unfolding-note
                          (cdr (assoc "UNFOLDING_PARENT"
                                      (manifolding-atlas-note-unfoldings unfolding-note))))))
    (if unfolding-parent
        ;; Scoped unique per parent — orphaned iff parent is the deleted file
        (let ((deleted-id (or my/manifolding-atlas-links--deleted-id
                              (ignore-errors
                                (with-temp-buffer
                                  (insert-file-contents deleted-file)
                                  (when (re-search-forward "^:ID:[ \t]+\\(.+\\)" nil t)
                                    (string-trim (match-string 1))))))))
          (and deleted-id (string= unfolding-parent deleted-id)))
      ;; Legacy shared — full scan
      (let* ((root (my/manifolding-atlas-root-dir))
            (deleted-exp (expand-file-name deleted-file))
            (unfoldings-dir (expand-file-name "admin/unfoldings/" (my/manifolding-atlas-root-dir)))
            (files (nconc (directory-files-recursively root "\\.org\\'")
                          (directory-files-recursively root "[^./]+\\'")))
            (orphaned t))
        (dolist (f files)
          (let ((exp (expand-file-name f)))
            (when (and (not (string-equal exp deleted-exp))
                       (not (file-in-directory-p exp unfoldings-dir)))
              (when (my/manifolding-atlas-links--unfolding-id-in-file-p unfolding-id exp)
                (setq orphaned nil)))))
        orphaned))))

(defun my/manifolding-atlas-links--unfolding-rel-path (unfolding-id)
  "Return relative path (from vault root) of the unfolding note UNFOLDING-ID."
  (when-let ((note (manifolding-atlas-db-get-by-id unfolding-id)))
    (let ((path (manifolding-atlas-note-path note)))
      (and path
           (file-relative-name path (my/manifolding-atlas-root-dir))))))
#+end_src

*** Missing-Links Guard

**** Open TODO counter
#+begin_src emacs-lisp
(defun my/manifolding-atlas-links--broken-links-open-count ()
  "Return count of open TODO entries in Missing Links.
Returns 0 if the file does not exist."
  (let ((path (my/manifolding-atlas-links--broken-links-path)))
    (if (not (file-exists-p path))
        0
      (with-temp-buffer
        (insert-file-contents path)
        (org-mode)
        (let ((count 0))
          (org-element-map (org-element-parse-buffer) 'headline
            (lambda (h)
              (when (equal (org-element-unfolding :todo-keyword h) "TODO")
                (setq count (1+ count)))))
          count)))))

**** Deletion guard for the queue file itself
#+begin_src emacs-lisp
(defun my/manifolding-atlas-links--guard-broken-links-file (file &rest _)
  "Before Missing Links is deleted: check for open TODO entries.
Warns and asks for confirmation if any remain unresolved."
  (let ((expanded (expand-file-name file)))
    (when (string= expanded (my/manifolding-atlas-links--broken-links-path))
      (let ((count (my/manifolding-atlas-links--broken-links-open-count)))
        (when (> count 0)
          (unless (y-or-n-p
                   (format "Missing Links has %d unresolved TODO%s. Delete anyway? "
                           count
                           (if (= count 1) "" "s")))
            (user-error "Deletion cancelled: %d unresolved broken link%s remain"
                        count
                        (if (= count 1) "" "s"))))))))

#+begin_src emacs-lisp
(defun my/manifolding-atlas-links--write-direction-todos
    (deleted-id deleted-title links peer-key direction)
  "Write a Missing Links TODO per LINK; PEER-KEY picks the other end.
Links originating from drawer unfolding entries, or touching notes
under admin/unfoldings/, route to Broken Unfoldings instead."
  (let ((deleted-path (my/manifolding-atlas-links--note-path-of deleted-id)))
    (dolist (link links)
      (let* ((peer-id (plist-get link peer-key))
             (peer-note (manifolding-atlas-db-get-by-id peer-id))
             (peer-title (if peer-note
                             (manifolding-atlas-note-title peer-note)
                           peer-id)))
        (cond
         ((eq peer-key :source)
          (let ((key (my/manifolding-atlas-links--drawer-unfolding-for
                      peer-id deleted-id)))
            (cond (key
                   (my/manifolding-atlas-links--write-broken-unfolding-todo
                    peer-id peer-title key deleted-title deleted-id direction))
                  ((my/manifolding-atlas-links--in-unfoldings-dir-p deleted-path)
                   (my/manifolding-atlas-links--write-broken-unfolding-todo
                    peer-id peer-title
                    (my/manifolding-atlas-links--family-of-path deleted-path)
                    deleted-title deleted-id direction))
                  (t
                   (my/manifolding-atlas-links--write-todo
                    peer-id peer-title deleted-title deleted-id direction)))))
         (t
          (let ((key (my/manifolding-atlas-links--drawer-unfolding-for
                      deleted-id peer-id)))
            (cond ((my/manifolding-atlas-links--in-unfoldings-dir-p deleted-path)
                   (my/manifolding-atlas-links--write-broken-unfolding-todo
                    deleted-id deleted-title
                    (my/manifolding-atlas-links--family-of-path deleted-path)
                    peer-title peer-id direction))
                  (key
                   (my/manifolding-atlas-links--write-broken-unfolding-todo
                    deleted-id deleted-title key peer-title peer-id direction))
                  (t
                   (my/manifolding-atlas-links--write-todo
                    deleted-id deleted-title peer-title peer-id direction))))))))))

(defun my/manifolding-atlas-links--read-blueprint-meta (file)
  "Return (KEY ID CONTEXTS KIND) from blueprint FILE, or nil without a KEY heading.
KEY matching is case-sensitive: only UPPER_SNAKE level-1 headings count.
CONTEXTS/KIND are read from plain tags on the KEY heading, e.g.
`* MY_KEY :unfolding:file:heading:'; ID still comes from the
heading's own :PROPERTIES: drawer, when present."
  (when (and file (file-exists-p file))
    (with-temp-buffer
      (insert-file-contents file)
      (org-mode)
      (goto-char (point-min))
      (let ((case-fold-search nil))
        (when (re-search-forward "^\\* \\([A-Z][A-Z0-9_]*\\)\\([ \t]+\\(:[A-Za-z0-9_:@]+:\\)\\)?[ \t]*$" nil t)
          (let* ((key (match-string-no-properties 1))
                 (tags (split-string (or (match-string-no-properties 3) "") ":" t))
                 (down (mapcar #'downcase tags))
                 (kind (car (cl-intersection
                             down
                             (mapcar #'symbol-name my/manifolding-atlas-blueprint-kinds)
                             :test #'string=)))
                 (contexts (string-join
                            (cl-intersection
                             down
                             (mapcar #'symbol-name my/manifolding-atlas-blueprint-contexts)
                             :test #'string=)
                            " ")))
            (goto-char (match-beginning 0))
            (list key
                  (org-entry-get nil "ID")
                  (unless (string-empty-p contexts) contexts)
                  kind)))))))

  "Remove drawer-order.org `**' headings matching no prompt key.
Compares against spec :keys plus the reinforce extra-keys allowlist
\(e.g. ID). Skips entirely when the specs registry is empty — a
failed registration must never eat the order file. Applies
\(setq + vault sweep) when anything was removed. Returns removed list."
  (let ((removed nil))
    (unless (and (boundp 'my/manifolding-atlas-org-prompt--specs)
                 (hash-table-p my/manifolding-atlas-org-prompt--specs)
                 (> (hash-table-count my/manifolding-atlas-org-prompt--specs) 0))
      (message "drawer-order: prune skipped (prompt registry unavailable)"))
    (when (and (fboundp 'my/manifolding-atlas-drawer-order-from-file)
               (fboundp 'my/manifolding-atlas-drawer-order-file)
               (boundp 'my/manifolding-atlas-org-prompt--specs)
               (hash-table-p my/manifolding-atlas-org-prompt--specs)
               (> (hash-table-count my/manifolding-atlas-org-prompt--specs) 0))
      (let ((known nil)
            (extra (and (boundp 'my/manifolding-atlas-reinforce--drawer-order-extra-keys)
                        my/manifolding-atlas-reinforce--drawer-order-extra-keys))
            (f (my/manifolding-atlas-drawer-order-file)))
        (maphash (lambda (_k spec)
                   (let ((key (plist-get spec :key)))
                     (when (and (stringp key) (not (string-empty-p key)))
                       (push key known))))
                 my/manifolding-atlas-org-prompt--specs)
        (let ((order (my/manifolding-atlas-drawer-order-from-file)))
          (setq removed (cl-remove-if
                         (lambda (k) (or (member k known) (member k extra)))
                         order)))
        (when (and removed (file-exists-p f))
          (with-temp-file f
            (insert-file-contents f)
            (dolist (k removed)
              (goto-char (point-min))
              (when (re-search-forward
                     (format "^\\*\\* +%s[ \t]*$" (regexp-quote k)) nil t)
                (delete-region (line-beginning-position)
                               (min (1+ (line-end-position)) (point-max))))))
          (when (fboundp 'my/manifolding-atlas--apply-drawer-order-file)
            (my/manifolding-atlas--apply-drawer-order-file))
          (message "drawer-order: pruned %d stale keys (%s)"
                   (length removed) (string-join removed ", ")))))
    removed))

(defun my/manifolding-atlas-links--reconcile-on-boot ()
  "Diff on-disk prompt keys vs `.known-keys' and reconcile.
New keys: drawer-order append + stub sweep (Added log per file).
Gone keys: Deleted log per holder file, static-sentinel drawer lines
stripped (live values kept + logged to Deleted Live Headings
Blueprints), Static Blueprints rebuilt. Stale drawer-order headings
(no prompt key) are pruned.
Updates `.known-keys' to current reality. Idempotent —
a second boot with no changes is a no-op. Returns (NEW . GONE)."
  (let ((current (my/manifolding-atlas-links--scan-current-keys))
        (last-known (my/manifolding-atlas-links--known-keys-read)))
    (if (null last-known)
        (progn
          (my/manifolding-atlas-links--known-keys-write current)
          (message "known-keys: seeded %d keys (no diff on first record)"
                   (length current))
          (cons nil nil))
      (let ((new (sort (cl-set-difference current last-known :test #'string=)
                       #'string<))
            (gone (sort (cl-set-difference last-known current :test #'string=)
                        #'string<))
            (vault-files (my/manifolding-atlas-links--vault-note-files))
            (cleanup-stripped 0)
            (cleanup-live 0))
        (when new
          (my/manifolding-atlas-links--ensure-drawer-order-keys new)
          (when (fboundp 'my/manifolding-atlas--sweep-sync-drawer-keys)
            (let ((specs (and (boundp 'my/manifolding-atlas-org-prompt--specs)
                              my/manifolding-atlas-org-prompt--specs)))
              (my/manifolding-atlas--sweep-sync-drawer-keys
               (mapcar (lambda (k)
                         (let ((spec (and (hash-table-p specs)
                                          (gethash (intern (downcase k)) specs))))
                           (cons k (if (and spec
                                            (fboundp 'my/manifolding-atlas-org-prompt--default-for-spec))
                                       (my/manifolding-atlas-org-prompt--default-for-spec spec)
                                     "TODO"))))
                       new)))))
        (dolist (k gone)
          ;; One bad key must never abort prune + known-keys rewrite below.
          (condition-case err
              (progn
                (dolist (f (my/manifolding-atlas-links--gone-key-holders
                            k vault-files))
                  (my/manifolding-atlas-links--log-removed-property f k))
                (when (fboundp 'my/manifolding-atlas--sweep-remove-drawer-key-if-static)
                  (let ((res (my/manifolding-atlas--sweep-remove-drawer-key-if-static k)))
                    (setq cleanup-stripped (+ cleanup-stripped (car res)))
                    (dolist (lv (cdr res))
                      (my/manifolding-atlas-links--write-deleted-live-property
                       k (car lv) (cdr lv)))
                    (setq cleanup-live (+ cleanup-live (length (cdr res)))))))
            (error (lwarn 'manifolding-atlas :warning
                          "known-keys: gone-key %s cleanup failed: %s"
                          k err))))
        (my/manifolding-atlas-links--prune-drawer-order-stale)
        (when gone
          (when (fboundp 'my/manifolding-atlas-missing--rebuild-from-health)
            (my/manifolding-atlas-missing--rebuild-from-health))
          (when (or (> cleanup-stripped 0) (> cleanup-live 0))
            (message "drawer cleanup: stripped %d static, preserved %d live (see Deleted Live Headings Blueprints)"
                     cleanup-stripped cleanup-live)))
        (my/manifolding-atlas-links--known-keys-write current)
        (when (or new gone)
          (message "known-keys: +%d -%d (%s / %s)"
                   (length new) (length gone)
                   (string-join new ", ") (string-join gone ", ")))
        (cons new gone)))))

(defun my/manifolding-atlas-links--check-direction
    (deleted-id deleted-title links verb peer-key direction)
  "Ask about DELETED-TITLE's LINKS (VERB phrase); record TODOs if confirmed."
  (let ((count (length links))
        (plural (if (= (length links) 1) "" "s")))
    (when (> count 0)
      (let ((names (mapcar (lambda (link)
                             (my/manifolding-atlas-links--peer-title link peer-key))
                           links)))
        (if (y-or-n-p
             (format "%s %s %d note%s: %s. Delete anyway? "
                     deleted-title verb count plural
                     (string-join names ", ")))
            (my/manifolding-atlas-links--write-direction-todos
             deleted-id deleted-title links peer-key direction)
          (user-error "Deletion cancelled: %s %s %d note%s"
                      deleted-title verb count plural))))))

**** Guard dispatcher
Two checks, each prompting separately with y-or-n-p. y — proceeds,
recording TODO entries for broken links. n — cancels deletion.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-links--guard-deletion (deleted-id deleted-title)
  "Guard deletion of DELETED-ID / DELETED-TITLE.
1. Incoming: notes that link TO this note (incl. wormholes).
2. Outgoing: notes that this note links TO."
  (when (and deleted-id deleted-title)
    ;; Check 1: incoming links and wormholes
    (let ((incoming (manifolding-atlas-db-query-links-to deleted-id)))
      (dolist (id (my/manifolding-atlas-links--wormhole-sources deleted-id))
        (push (list :source id) incoming))
      (my/manifolding-atlas-links--check-direction
       deleted-id deleted-title incoming "is linked from" :source "incoming"))
    ;; Check 2: outgoing links
    (let ((outgoing (manifolding-atlas-db-query-links-from deleted-id)))
      (my/manifolding-atlas-links--check-direction
       deleted-id deleted-title outgoing "links to" :dest "outgoing"))))

*** Pre-Delete Capture: File

**** Eligibility predicate
Skips admin files, unfolding files, and non-org files.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-links--deletion-guardable-p (expanded)
  "Non-nil when EXPANDED is a deletable vault note file."
  (and (file-exists-p expanded)
       (file-in-directory-p expanded (my/manifolding-atlas-root-dir))
       (not (file-in-directory-p
             expanded
             (expand-file-name "admin/unfoldings/" (my/manifolding-atlas-root-dir))))
       (or (string-suffix-p ".org" expanded)
            (not (string-match-p "\\." (file-name-nondirectory expanded))))
       ;; Compare full paths: the admin vars hold "admin/..." relative
       ;; paths, so basename comparison never matched and admin files
       ;; (e.g. Missing Links deleted when empty) got logged as junk.
       (not (member (expand-file-name expanded)
                    (my/manifolding-atlas-links--admin-file-paths)))))
#+end_src

**** Before-file-delete advice
Guards the queue file, reads ID/title, runs the link guard, captures
orphaned unfoldings while the file still exists, then records the entry.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-links--before-file-delete (file &rest _)
  "Before FILE is deleted: guard, capture state, record."
  (let ((expanded (expand-file-name file)))
    (when (my/manifolding-atlas-links--deletion-guardable-p expanded)
      ;; Guard broken-links specifically
      (my/manifolding-atlas-links--guard-broken-links-file expanded)
      ;; Read ID and title then run link guard
      (with-temp-buffer
        (insert-file-contents expanded)
        (org-mode)
        (let (id title)
          (setq id    (org-entry-get (point-min) "ID"))
          (setq title (cadr (assoc "TITLE"
                                   (org-collect-keywords
                                    (list "TITLE")))))
          (when id
            (setq my/manifolding-atlas-links--deleted-id    id
                  my/manifolding-atlas-links--deleted-title
                  (or title (file-name-base expanded)))
            (my/manifolding-atlas-links--guard-deletion
             id
             my/manifolding-atlas-links--deleted-title)))
        ;; Capture unfoldings orphaned by this deletion while the file exists
        (setq my/manifolding-atlas-links--deleted-unfoldings
              (my/manifolding-atlas-links--orphaned-unfoldings-for expanded))
        ;; Record in ReManifold even without ID
        (my/manifolding-atlas-links--write-deleted-entry
         expanded my/manifolding-atlas-links--deleted-unfoldings)))))

(defun my/manifolding-atlas-links--after-file-delete (&rest _)
  "After file deletion: delete orphaned unfoldings and clear captured state.
Unfoldings are captured into a local first so the delete-file advice firing
again for each unfolding cannot re-enter with stale state."
  (let ((unfoldings my/manifolding-atlas-links--deleted-unfoldings))
    (setq my/manifolding-atlas-links--deleted-id    nil
          my/manifolding-atlas-links--deleted-title nil
          my/manifolding-atlas-links--deleted-unfoldings  nil)
    (dolist (rel unfoldings)
      (let ((path (expand-file-name rel (my/manifolding-atlas-root-dir))))
        (when (file-exists-p path)
          (delete-file path))))))

(defun my/manifolding-atlas-links--before-heading-delete (&rest _)
  "Before a heading is cut or archived: capture ID and title,

        my/manifolding-atlas-links--deleted-title nil))

(defun my/manifolding-atlas-links-setup ()
  "Register all advice hooks for the links guard.

  ;; File deletion

  (advice-add 'org-cut-subtree

              :after  #'my/manifolding-atlas-links--after-heading-delete)
  ;; Heading archive

              :before #'my/manifolding-atlas-links--before-heading-delete)

              :after  #'my/manifolding-atlas-links--after-heading-delete))

(my/manifolding-atlas-links-setup)

(defun my/manifolding-atlas-links-audit ()
  "Scan entire vault for dead id: links using manifolding-atlas native API.
Produces *manifolding-atlas-dead-links* buffer. Run after bulk operations."
  (interactive)
  (let ((dead (manifolding-atlas-db-query-dead-links)))
    (if (not dead)
        (message "No dead links found in vault.")
      (with-current-buffer (get-buffer-create "*manifolding-atlas-dead-links*")
        (erase-buffer)
        (insert (format "Dead links: %d\n\n" (length dead)))
        (pcase-dolist (`(,note . ,dest-id) dead)
          (insert (format "  [%s]  %s  ->  missing id: %s\n"
                          (manifolding-atlas-note-id note)
                          (manifolding-atlas-note-title note)
                          dest-id)))
        (display-buffer (current-buffer))))))
