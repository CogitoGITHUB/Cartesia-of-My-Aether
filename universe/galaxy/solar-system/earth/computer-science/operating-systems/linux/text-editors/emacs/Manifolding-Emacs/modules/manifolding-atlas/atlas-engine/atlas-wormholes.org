#+begin_src emacs-lisp
(defun wormhole-clone-set-overlays (&rest overlays)
  "Add wormhole-clone properties to OVERLAYS.
This function directly modifies the OVERLAYS passed and returns
modified OVERLAYS as a list in the same order as passed to it.
No copy is produced.

This function also refreshes `wormhole-clone-overlays' as a
side-effect to keep track of the current wormhole-clone overlays,
which is primarily used to clean up wormhole-clone overlays with
`wormhole-clone-delete-overlays'.

This function does not explicitly differentiate overlays for the
original text region and its clones.  Where such distinction is
important, use the sequence of OVERLAYS list; for example, the
first element of the list can be the overlay for the original and
rest, clones.

As this function returns modified overlays, a calling function
can further modify them.  For instance, you can put different
faces to visually differentiate them."
  (if (or (not overlays)
          (> 2 (length overlays)))
      (user-error "Nothing done.  You need to pass 2 or more overlays")
    (setq wormhole-clone-overlays nil)
    (dolist (ov overlays)
      (overlay-put ov 'evaporate t)
      (overlay-put ov 'wormhole-clones overlays)
      (overlay-put ov 'modification-hooks
                   '(wormhole-clone-live-sync))
      (overlay-put ov 'insert-in-front-hooks
                   '(wormhole-clone-live-sync))
      (overlay-put ov 'insert-behind-hooks
                   '(wormhole-clone-live-sync))
      (overlay-put ov 'priority -50)
      ;; Add a local post-command-hook for each overlay buffer
      (with-current-buffer (overlay-buffer ov)
        (add-hook 'post-command-hook #'wormhole-clone-post-command-h nil t)))
    (setq wormhole-clone-overlays overlays)
    overlays))


#+end_src

#+begin_src emacs-lisp
(defun wormhole-clone-post-command-h ()
  "Delete all the wormhole-clone overlays when any one is non-existent."
  (when-let ((ovs wormhole-clone-overlays))
    (let ((deleted nil))
      (dolist (ov ovs)
        (unless (or deleted
                    (overlay-buffer ov))
          (setq deleted t)))
      (when deleted
        (wormhole-clone-delete-overlays)))))


#+end_src

#+begin_src emacs-lisp
(defun wormhole-clone-delete-overlays ()
  "Remove all live-sync overlays.
Return a list of the buffer, beginning and ending points of the
deleted overlays.  Each element of the list is in this structure:

    (buf (beg . end))

This function checks `wormhole-clone-overlays' for overlays being
tracked.  Return nil if there is no overlay in it.

As side-effects, this function also does the following to clean
up wormhole-clone:

- Remove the local `post-command-hook'
  `wormhole-clone-post-command-h' for wormhole-clone each overlay

- Reset tracking of wormhole-clone overlays by setting
  `wormhole-clone-overlays' to nil"
  (when wormhole-clone-overlays
    (let ((ovs wormhole-clone-overlays)
          deleted-overlays)
      (dolist (ov ovs)
        ;; Clean up the local post-command-hook
        (let ((element (list (overlay-buffer ov) (cons (overlay-start ov) (overlay-end ov)))))
          (push element deleted-overlays)
          (when (overlay-buffer ov)
            (with-current-buffer (overlay-buffer ov)
              (remove-hook 'post-command-hook
                           #'wormhole-clone-post-command-h t)

          (delete-overlay ov)))))
      (setq wormhole-clone-overlays nil)
      ;; As push is used to construct the list, the sequence needs to be reversed
      (nreverse deleted-overlays))))

#+begin_src emacs-lisp
(defun wormhole-clone-live-sync (ol1 after beg end &optional _len)
  "Propagate the change made under the overlay OL1 to the other paired clone.
This is used on the `modification-hooks' property of text clones.
AFTER, BEG, and END are the fixed args for `modification-hooks'
and friends in an overlay.

It's a simplified version of the original `wormhole-clone--maintain'.
This function does not use SPREADP or SYNTAX (both defined in
`wormhole-clone-create').

Overlay is also assumed to be always SPREADP but insteaf we opt
for (nil t) -- refer to `wormhole-clone-make-overlay'.  This enables
tighter overlay size and has `post-command-hook' to deal with the
case when one of the overlays is deleted (refer to
`wormhole-clone-post-command-h').

This function also works during undo in progress; that is, when
`undo-in-progress' is non-nil."
  (when (and after
             (not wormhole-clone-live-sync-in-progress)
             (overlay-start ol1)
             (<= beg end))
    (save-excursion
      ;; Now go ahead and update the clones.
      (let ((head (- beg (overlay-start ol1)))
            (tail (- (overlay-end ol1) end))
            (str (buffer-substring-no-properties beg end)) ;changed to no-properties
            (wormhole-clone-live-sync-in-progress t))
        (dolist (ol2 (overlay-get ol1 'wormhole-clones))
          (with-current-buffer (overlay-buffer ol2)
            (save-restriction
              (widen)
              (let ((oe (overlay-end ol2)))
                (unless (or (eq ol1 ol2) (null oe))
                  (let ((mod-beg (+ (overlay-start ol2) head)))
                    (goto-char (- oe tail))
                    (if (not (> mod-beg (point)))
                        (progn
                          (save-excursion (insert str))
                          (delete-region mod-beg (point)))
                      (user-error "No live-sync done.  \
The text strings in the overlays are not identical"))))))))))))

(defmacro atlas-wormholes-with-inhibit-read-only (&rest body)
  "Run BODY with `'inhibit-read-only` t.
This macro is used instead of `with-silent-modifications' because
Org mode's caching relies upon modification hooks to function."
  (declare (debug t) (indent 0))
  (let ((modified (make-symbol "modified")))
    `(let* ((,modified (buffer-modified-p))
            (inhibit-read-only t))
       (unwind-protect
           (progn
             ,@body)
         (unless ,modified
           (restore-buffer-modified-p nil))))))

(defun atlas-wormholes-make-from-link (&optional arg)
  "Make a wormhole keyword from a link at point.

The resultant wormhole keyword will be placed in the first
next empty line.  If there is no empty line until the bottom of
the buffer, this function adds a new empty line.

When minor-mode `atlas-wormholes-mode' is active, this function
automatically wormholes the text content; when it is inactive,
it simply adds the \"#+wormhole\" keyword before the link and
inserts the whole line.

If you pass a `universal-argument', this function reverses this:
if the mode is active, the keyword gets inserted; if the mode is
inactive, the wormhole gets added.

You can pass a prefix argument (ARG) with using
`digit-argument' (e.g. C-1 or C-2; or \\[universal-argument] 3,
so on) or `universal-argument' (\\[universal-argument]).

If you pass a positive number 1-9 with `digit-argument', this
function automatically puts the :level property to the resultant
wormhole keyword."
  (interactive "P")
  (let* ((context (org-element-lineage
                   (org-element-context) '(link) t))
         (auto-wormhole-p (if (or (not arg) (numberp arg))
                                atlas-wormholes-mode
                              ;; if `universal-argument' is passed,
                              ;; reverse nil/t when
                              (not atlas-wormholes-mode))))
    (let* ((contents-beg (org-element-property :contents-begin context))
           (contents-end (org-element-property :contents-end context))
           (contents (and contents-beg
                          (buffer-substring-no-properties contents-beg
                                                          contents-end)))
           (link (org-element-link-interpreter context contents)))
      (save-excursion
        (atlas-wormholes-search-or-add-next-empty-line)
        (insert (format "#+wormhole: %s\n" link))
        (forward-line -1)
        (when (and (numberp arg)
                   (> arg 0)
                   (<= arg 9))
          (end-of-line)
          (insert (format " :level %d" arg)))
        (when auto-wormhole-p (atlas-wormholes-add))))))

(defun atlas-wormholes-insert (&optional insert-above)
  "Insert #+WORMHOLE: from a link at point or new link to a blank line.
If the point is not at a link but has text, this command will try to
find the first link in the current line. If the point is at a blank
line, this command will call a function in set to variable

          (or (org-element-lineage (org-element-context) 'link t) ; at-point

      (unless blank-line-p

        (unless (bolp) (newline nil t)))

       (point) (line-end-position) 'pulse-highlight-start-face))))

(defun atlas-wormholes-add (&optional copy)
  "Wormhole text content for the #+wormhole at point.
When minor-mode `atlas-wormholes-mode' is inactive in the
current buffer, this function toggles it on.

With using `universal-argument' (\\[universal-argument]) or
non-nil COPY argument, you can copy the wormholed content into
the buffer instead of wormhole.

Examples of acceptable formats are as below:

- \"#+wormhole: [[file:path/file.org::search-option][desc]]:level n\"
- \"#+wormhole: [[id:uuid]] :level n :only-contents\"

The file path or id in the wormhole keyword value are
translated to the normal Org Mode link format such as
[[file:path/to/file.org::*Heading]] or [[id:uuid]] to copy a piece
of text from the link target.

TODO: id:uuid without brackets [[]] is a valid link within Org
Mode. This is not supported yet.

A wormholed text region is read-only. You can use a variety of
commands on the wormholed region at point. Refer to the
commands below. You can customize the keymap with
using `atlas-wormholes-map'.

For example, `atlas-wormholes-live-sync-start' lets you edit the
part of the text at point.  This edit mode is analogous to Occur
Edit for Occur Mode.

TODO: that for wormholes of Org elements/buffer, live-sync
does not support all the elements.

\\{atlas-wormholes-map}"
  (interactive "P")
  (when (atlas-wormholes-check-add)
    ;; Turn on the minor mode to load extensions before staring to add.
    (unless atlas-wormholes-mode
      (let ((atlas-wormholes-add-all-on-activate nil))
        (atlas-wormholes-mode +1)))
    (let* ((keyword-plist (atlas-wormholes-keyword-string-to-plist))
           (link (atlas-wormholes-wrap-path-to-link
                  (plist-get keyword-plist :link)))
           ;; Note 2025-01-03 Retrospectively, PAYLOAD feels redundant now that
           ;; `atlas-wormholes-add' is being refactored. For
           ;; backword-compatibility, I am keeping PAYLOAD.
           (payload (run-hook-with-args-until-success
                     'atlas-wormholes-add-functions link keyword-plist))
           (tc-type (plist-get payload :tc-type))
           (content (plist-get payload :src-content))
           (keyword-plist (if (atlas-wormholes-type-is-org tc-type)
                              (plist-put
                               keyword-plist :current-level
                               (or (org-current-level) 0))
                            keyword-plist))
           (content
            (run-hook-with-args-until-success
             'atlas-wormholes-content-format-functions
             tc-type content keyword-plist)))
      (cond ((functionp payload)
             ;; Allow for asynchronous wormhole
             (funcall payload link keyword-plist copy))
            ;; No content
            ((or (string-empty-p content)
                 (eq content nil))
             ;; Keep going with program when no content `atlas-wormholes-add-all'
             ;; should move to the next wormhole
             (prog1 nil
               (message
                "No content found with \"%s\".  Check the link at point %d, line %d"
                (org-element-property :raw-link link)
                (point) (org-current-line))))
            ;; Normal case
            (t
             (pcase-let* ((org-src-tab-acts-natively (if (not atlas-wormholes-src-tab-acts-natively)
                                                         nil
                                                       org-src-tab-acts-natively))
                          (buffer-modified-p (buffer-modified-p))
                          (`(,beg . ,end) (atlas-wormholes-content-insert content)))
               (when (and beg end)
                 ;; If COPY, then we want to get the buffer-modified-p flag always t.
                 ;; If wormhole, it should be the same as before wormhole.
                 (if copy
                     (restore-buffer-modified-p t)
                   (restore-buffer-modified-p buffer-modified-p)
                   (with-silent-modifications
                     (atlas-wormholes-content-add-text-props-and-overlay
                      payload keyword-plist beg end)))
                 (run-hook-with-args 'atlas-wormholes-after-add-functions
                                     beg end))))))))

(defun atlas-wormholes-add-all (&optional narrowed)

(defun atlas-wormholes-remove ()
  "Remove wormholed text at point.
When success, return the beginning point of the keyword re-inserted."
  (interactive)
  (unless (atlas-wormholes-within-wormhole-p)
    (user-error "Nothing done. No wormhole exists here."))
  (when-let*
      ((beg-end (plist-get (atlas-wormholes-at-point) :location))
       (beg (car beg-end))
       (end (cdr beg-end))
       (keyword-plist (get-char-property (point)
                                         'atlas-wormholes-orig-keyword))
       (indent (plist-get keyword-plist :current-indentation))
       (keyword (atlas-wormholes-keyword-plist-to-string keyword-plist))
       (tc-pair-ov (get-char-property (point) 'atlas-wormholes-pair)))
    (prog1
        beg
      (let ((buffer-modified-p (buffer-modified-p)))
        (when (atlas-wormholes-within-live-sync-p)
          (atlas-wormholes-live-sync-exit))
        ;; Clean up source buffer fringe indicators before deleting overlay
        (when (overlay-buffer tc-pair-ov)
          (let ((src-buf (overlay-buffer tc-pair-ov))
                (src-beg (overlay-start tc-pair-ov))
                (src-end (overlay-end tc-pair-ov)))
            ;; Remove our fringe indicators from source
            (atlas-wormholes-remove-fringe-from-region src-buf src-beg src-end)
            ;; Run hooks for extensions to do additional cleanup
            (run-hook-with-args 'atlas-wormholes-after-remove-functions
                                src-buf src-beg src-end)))
          (delete-overlay tc-pair-ov)
          ;; Careful with the point (beg and end). `save-excursion' may work but
          ;; since we are removing a region and inserting the keyword back, it is
          ;; safer to explicitly go back to beg with `goto-char'. Positions are
          ;; especially important when two wormholes are present consecutively
          ;; without space in-between.
          (atlas-wormholes-with-inhibit-read-only
            (delete-region beg end)
            (goto-char beg) (insert-before-markers keyword))
          (goto-char beg) (when (> indent 0) (indent-to indent))
          ;; Removing wormhole should not be considered buffer modification.
          ;; Keep the buffer-modified-p flag as it was before removal.
          (restore-buffer-modified-p buffer-modified-p)))))

(defun atlas-wormholes-live-sync-start ()
  "Start live-sync edit on the wormhole at point.

While live-sync is on, before- and after-save-hooks to remove/add
wormholes are also temporarily disabled.  This prevents
auto-save from getting in the way of live-sync.

For wormholes of Org elements or a buffer, live-sync works
only on the following elements: center-block, drawer,
dynamic-block, latex-environment, paragraph, plain-list,
quote-block, special-block table, and verse-block.

It is known that live-sync does not work for the other Org
elements: comment-block, export-block, example-block,
fixed-width, keyword, src-block, and property-drawer.

`atlas-wormholes-live-sync-map' inherits `org-mode-map' and adds
a couple of atlas-wormholes specific keybindings; namely:

- `atlas-wormholes-live-sync-paste'
- `atlas-wormholes-live-sync-exit'

\\{atlas-wormholes-live-sync-map}

TODO: Support asynchronous wormholes."
  (interactive)
  (if (not (atlas-wormholes-within-wormhole-p))
      (progn (message (format "Nothing done. Not a translusion at %d" (point)))
             nil)
    ;; Delete the other live-sync and refresh its wormhole
    ;; There should be only one pair of wormhole-source in live-sync
    (when-let* ((deleted-live-sync-ovs (wormhole-clone-delete-overlays))
                (deleted-tc-ov (cadr deleted-live-sync-ovs)))
      (atlas-wormholes-live-sync-refresh-after-exit deleted-tc-ov))
    (atlas-wormholes-refresh)
    (let* ((remember-pos (point))
           (ovs (atlas-wormholes-live-sync-buffers))
           (src-ov (car ovs))
           (tc-ov (cdr ovs))
           (tc-beg (overlay-start tc-ov))
           (tc-end (overlay-end tc-ov)))
      ;; Check the length of both overlays
      ;; if different, abort live-sync
      (if (not (= (- (overlay-end tc-ov) (overlay-start tc-ov))
                  (- (overlay-end src-ov) (overlay-start src-ov))))
          (progn
            (user-error
             (concat
              "No live-sync can be started.  "
              "Lengths of wormhole and source are not identical"
              (format " - tc: [%s] src: [%s]"
                      (- (overlay-end tc-ov) (overlay-start tc-ov))
                      (- (overlay-end src-ov) (overlay-start src-ov)))))
            nil) ; return nil
        (atlas-wormholes-live-sync-modify-overlays
         (wormhole-clone-set-overlays src-ov tc-ov))
        (atlas-wormholes-live-sync-display-buffer (overlay-buffer src-ov))
        (goto-char remember-pos)
        (remove-hook 'before-save-hook #'atlas-wormholes-before-save-buffer t)
        (remove-hook 'after-save-hook #'atlas-wormholes-after-save-buffer t)
        (with-silent-modifications
          (remove-text-properties (1- tc-beg) tc-end '(read-only)))
        t))))

(defun atlas-wormholes-add-source-marker (link)
  "Return the marker of wormhole source by opening LINK.
LINK must be Org's link object that `org-link-open' can act on. As long
as `org-link-open' opens a buffer within Emacs, this function should
return a marker.

When LINK has no search option (::*Heading, ::CUSTOM_ID, etc.), returns
marker at 1 to ensure line-based wormholes calculate
offsets from file beginning, not from cursor position."
  ;; Assume the point now is the wormholing buffer

  ;; Note 2025-12-18 `org-link-open' does not necessarily obey
  ;; `display-buffer-alist' and can open the target buffer in the currently
  ;; selected window. This is disruptive for users. We want wormholes to
  ;; keep the current buffer in the current window. To do this, it seems
  ;; `save-window-excursion' is the only way.

  ;; Note 2026-03-07 For `org-link-open', instead of relying on
  ;; `save-window-excursion', we are now using display-buffer-alist with its
  ;; ACTION `display-buffer-no-window' and allow-no-window being non-nil. This
  ;; is originally conceived as a fix for GitHub issue #299, but it looks like
  ;; it can be a good way to tame `org-link-open' for our purpose.

  ;; This `save-excursion' is needed for the case where the target and
  ;; source are the same buffer.
  (when (string= "info" (org-element-property :type link))
        (user-error
         "Atlas-wormholes: wormholing info node is currently unsupported."))
  (save-excursion
    (condition-case nil
        (let (
              ;; Don't ever prompt to create a headline when wormholing.
              ;; t is a less surprising default than nil - fuzzy search.
              (org-link-search-must-match-exact-headline t)
              ;; Don't display or pop to the link's target buffer. Non-nil value
              ;; for allow-no-window will cause `pop-to-buffer' to fall back to
              ;; `set-buffer', which we want here.
              (display-buffer-alist '((".*" display-buffer-no-window
                                       (allow-no-window t)))))
          (org-link-open link)
          ;; In the target buffer temporarily.
          (save-excursion
            (move-marker
             (make-marker)
             (if (atlas-wormholes-source-point-beginning-of-buffer-p link)
                 1 (point)))))
      (error (user-error
              "Atlas-wormholes: `org-link-open' cannot open link, %s"
              (org-element-property :raw-link link))))))

  - No search option

(defun atlas-wormholes--fringe-spec-p (prop-value)
  "Return non-nil if UNFOLDING-VALUE represents a wormhole fringe indicator.
Checks both graphical fringe (display property) and
terminal fringe (face property)."
  (or
   ;; Graphical: (left-fringe BITMAP FACE)
   ;; BITMAP can be 'empty-line (source) or 'atlas-wormholes-fringe-bitmap (destination)
   (and (listp prop-value)
        (eq (car prop-value) 'left-fringe)
        (memq (cadr prop-value) '(empty-line atlas-wormholes-fringe-bitmap))
        (memq (nth 2 prop-value)
              '(atlas-wormholes-source-fringe
                atlas-wormholes-fringe)))
   ;; Terminal: face property with our face names
   (memq prop-value
         '(atlas-wormholes-source-fringe
           atlas-wormholes-fringe
           atlas-wormholes-source
           atlas-wormholes))))

#+begin_src emacs-lisp
(defun atlas-wormholes-prefix-has-fringe-p (prefix)
  "Return non-nil if PREFIX string contains a wormhole fringe indicator.
Checks for both graphical fringe (display property) and terminal
fringe (face property within the string)."
  (when (stringp prefix)
    (let ((pos 0))
      (catch 'found
        ;; Check display properties (graphical fringe)
        (while (setq pos (next-single-property-change pos 'display prefix))
          (when (atlas-wormholes--fringe-spec-p
                 (get-text-property pos 'display prefix))
            (throw 'found t)))

        ;; Check face properties within the string (terminal fringe only)
        ;; Terminal fringes are strings like "| " with face property
        (unless (display-graphic-p)
          (setq pos 0)
          (while (< pos (length prefix))
            (let ((face (get-text-property pos 'face prefix)))
              (when (memq face '(atlas-wormholes-source-fringe
                                 atlas-wormholes-fringe))
                (throw 'found t)))
            (setq pos (1+ pos))))

        nil))))

(defun atlas-wormholes-append-fringe-to-prefix (existing-prefix face)
  "Append fringe indicator to EXISTING-PREFIX, preserving it.
FACE determines the fringe color (atlas-wormholes-source-fringe or
atlas-wormholes-fringe).
Returns concatenated string suitable for `line-prefix' or `wrap-prefix'.

In terminal mode, prepends fringe to place it at the left margin.
In graphical mode, appends fringe to preserve indentation alignment."

    (if existing-prefix
        (if (display-graphic-p)

#+begin_src emacs-lisp
(defun atlas-wormholes-add-fringe-to-region (buffer beg end face)
  "Add fringe indicator to each line in BUFFER between BEG and END.
FACE determines the fringe color.

When org-indent-mode is active (`line-prefix'/`wrap-prefix' properties exist),
appends fringe to existing indentation. When org-indent-mode is inactive,
adds fringe-only prefix."
  (with-current-buffer buffer
    (with-silent-modifications
      (save-excursion
        (goto-char beg)
        (catch 'done
          (while (< (point) end)
            (let* ((line-beg (line-beginning-position))
                   (line-end (min (line-end-position) end))
                   (line-prefix (get-text-property line-beg 'line-prefix))
                   (wrap-prefix (get-text-property line-beg 'wrap-prefix))
                   (fringe-only (atlas-wormholes--make-fringe-indicator face)))

              ;; Handle line-prefix
              (if line-prefix
                  ;; org-indent-mode case: append to existing prefix
                  (unless (atlas-wormholes-prefix-has-fringe-p line-prefix)
                    (atlas-wormholes--update-line-prefix
                     line-beg line-end 'line-prefix
                     (atlas-wormholes-append-fringe-to-prefix line-prefix face)))
                ;; Non-indent case: add fringe-only prefix
                (atlas-wormholes--update-line-prefix
                 line-beg line-end 'line-prefix fringe-only))

              ;; Handle wrap-prefix
              (if wrap-prefix
                  ;; org-indent-mode case: append to existing prefix
                  (unless (atlas-wormholes-prefix-has-fringe-p wrap-prefix)
                    (atlas-wormholes--update-line-prefix
                     line-beg line-end 'wrap-prefix
                     (atlas-wormholes-append-fringe-to-prefix wrap-prefix face)))
                ;; Non-indent case: add fringe-only prefix
                (atlas-wormholes--update-line-prefix
                 line-beg line-end 'wrap-prefix fringe-only)))

            ;; Try to advance to next line; if we can't, we're done
            (when (not (zerop (forward-line 1)))
              (throw 'done nil))))))))

#+begin_src emacs-lisp
(defun atlas-wormholes-remove-fringe-from-prefix (prefix)
  "Remove fringe indicator from PREFIX string.
Returns the cleaned prefix, or nil if prefix was only the fringe indicator."
  (when (stringp prefix)
    (let* ((cleaned prefix)
           (pos (length cleaned)))
      ;; Remove all fringe indicators (both graphical and terminal)
      (while (>= pos 0)
        (let ((display-prop (get-text-property pos 'display cleaned))
              (face-prop (get-text-property pos 'face cleaned)))
          (when (or (atlas-wormholes--fringe-spec-p display-prop)
                    (atlas-wormholes--fringe-spec-p face-prop))
            ;; Found a fringe indicator, remove it
            (setq cleaned (concat (substring cleaned 0 pos)
                                  (substring cleaned (1+ pos)))))
          ;; Adjust position since we removed a character
          (setq pos (1- pos))))
      ;; Return nil if nothing left, otherwise return cleaned prefix
      (if (string-empty-p cleaned) nil cleaned))))

#+begin_src emacs-lisp
(defun atlas-wormholes-remove-fringe-from-region (buffer beg end)
  "Remove fringe indicators from each line in BUFFER between BEG and END.
This restores `line-prefix' and `wrap-prefix' to their state before
`atlas-wormholes-add-fringe-to-region' was called.

In `org-mode' buffers, removes only the fringe portion while preserving
org-indent indentation.  In non-org buffers, removes the properties
entirely since they were added solely for fringe display."
  (with-current-buffer buffer
    (with-silent-modifications
      (save-excursion
        (goto-char beg)
        (let ((is-org-buffer (derived-mode-p 'org-mode)))
          (while (< (point) end)
            (let* ((line-beg (line-beginning-position))
                   (line-end (min (line-end-position 1) end))
                   (line-prefix (get-text-property line-beg 'line-prefix))
                   (wrap-prefix (get-text-property line-beg 'wrap-prefix)))

              (if is-org-buffer
                  ;; Org buffer: strip fringes, preserve org-indent content
                  (progn
                    (when line-prefix
                      (atlas-wormholes--update-line-prefix
                       line-beg line-end 'line-prefix
                       (atlas-wormholes-remove-fringe-from-prefix line-prefix)))
                    (when wrap-prefix
                      (atlas-wormholes--update-line-prefix
                       line-beg line-end 'wrap-prefix
                       (atlas-wormholes-remove-fringe-from-prefix wrap-prefix))))
                ;; Non-org buffer: remove properties entirely
                (progn
                  (atlas-wormholes--update-line-prefix line-beg line-end 'line-prefix nil)
                  (atlas-wormholes--update-line-prefix line-beg line-end 'wrap-prefix nil))))
            (forward-line 1)))))))

(defun atlas-wormholes-find-source-marker (beg end)
  "Return marker that points to source begin point for wormhole.
It works on the wormhole region at point.  BEG and END are
meant to be wormhole region's begin and end used to limit the
`text-property-search' -- as it does not have an argument to
limit the search, this is done by looking at the output point and
compare it with BEG and END.

Return nil when :parent text-prop cannot be found.

This function critically relies on the fact that `org-element'
puts a \":parent\" text property to the elements obtained by
using `org-element-parse-buffer' and
`org-element--parse-elements' Some elements such as comment-block
does not seem to add :parent, which makes live-sync not working
for them.

Text properties are added by `org-element-put-property' which in
turn uses `org-add-props' macro. If any of this substantially
changes, the logic in this function will need to reviewed."
  (let ((parent (get-text-property (point) ':parent))
        (src-buf (overlay-buffer
                  (get-text-property (point) 'atlas-wormholes-pair)))
        (m))
    (unless parent
      (save-excursion
        (when-let ((match (or (text-property-search-forward
                               ':parent)
                              (text-property-search-backward
                               ':parent))))
          ;; Point must be between beg and end (inclusive)
          (when (and (<= beg (point)) (<= (point) end))
            (setq parent (prop-match-value match))))))
    (when parent
      (setq m (set-marker (make-marker)
                          (or (org-element-property :contents-begin parent)
                              (org-element-property :begin parent))
                          src-buf)))
    m))

(defun atlas-wormholes-at-point (&optional point)
  "Return plist representing the wormhole at point.
This function returns a plist of this form:

   (:id ID-STRING :location (BEG . END))

With Elisp, POINT can be passed. Otherwise, the current point is
used."
  (save-excursion
    (and-let* ((pt (or point (point)))
               ;; If the ID is present, the current point is within a
               ;; wormhole.
               (id (get-text-property pt 'atlas-wormholes-id))
               ;; We need to get both BEGINNING and END of the wormhole at
               ;; point. `prop-match-forward' sets BEGINNING as the current
               ;; point, rather than the beginning of the current wormhole,
               ;; so `prop-match-backward' is also used.
               (prop-match-forward
                (text-property-search-forward 'atlas-wormholes-id))
               ;; Because the cursor (or POINT) is unlikely to be at the
               ;; beginning, find the END point first.
               (end (prop-match-end prop-match-forward))
               (value (prop-match-value prop-match-forward))
               (prop-match-backward
                ;; As the call to `text-property-search-backward' needs to match
                ;; VALUE, t needs to be passed to PREDICATE unlike
                ;; `text-property-search-forward' a few lines above.
                (text-property-search-backward 'atlas-wormholes-id value t))
               (beg (prop-match-beginning prop-match-backward)))
      (list :id id :location (cons beg end)))))

  "Return an enclosing Org element between BEG and END.
This function is intended for live-sync.

This function first looks for elements other than paragraph:

If none of them found, this function identifies the paragraph at
point to return.

Note that live-sync is known to work only for the following elements:
  center-block drawer dynamic-block latex-environment plain-list
  quote-block special-block table verse-block

         (pos (point)))

        (delay-mode-hooks (org-mode))

                                          quote-block special-block
                                          src-block
                                          table
                                          verse-block) 'with-self)

                   (org-element-lineage
                    (org-element-context) '(paragraph) 'with-self))))
          (if context context
            (user-error (format "Live sync cannot start here: point %d"
                                (point)))))))))

(defun my/manifolding-atlas-extract-wormhole-links (ctx note-data)
  "Extract id: links from #+wormhole: keywords in CTX."
  (let* ((ast     (manifolding-atlas-parse-ctx-ast ctx))
         (note-id (plist-get note-data :id))
         (found   nil))
    (org-element-map ast 'keyword
      (lambda (kw)
        (when (string= (org-element-property :key kw) "WORMHOLE")
          (let ((value (org-element-property :value kw)))
            (when (string-match "\\[\\[id:\\([^] \t\n]+\\)\\]\\[" value)
              (push (match-string-no-properties 1 value) found))))))
    (when found
      (let ((unique (delete-dups (nreverse found))))
        (emacsql (manifolding-atlas-db)
                 [:delete :from wormhole_links :where (= source $s1)]
                 note-id)
        (emacsql (manifolding-atlas-db)
                 [:insert :into wormhole_links :values $v1]
                 (mapcar (lambda (d) (vector note-id d)) unique))
        (plist-put note-data :wormhole-links unique)))
    note-data))
