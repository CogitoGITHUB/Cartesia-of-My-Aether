#+title: Manifolding Atlas Routines Plugin
#+priority: 16

Modular routines system for the atlas, modeled on the mastering plugin.
A routine is an atlas note tagged =routine= whose child headings are
routine entries referencing sources via =SOURCE=, =ROUTINE_ID= and
=ROUTINE_ORDER= unfoldings. Every mutation dual-writes the org drawer
and the SQLite index, so queries run against the DB without re-parsing
org. Chains are routines whose sources are themselves routines.

*** Database Tables
Three tables: =routines= indexes routine notes by type/domain/temporal,
=routine_entries= maps entry headings to their parent routine with
ordering, and =routine_sessions= journals each run with done/skip counts.

**** Entries column migration
=CREATE IF NOT EXISTS= never adds columns to an existing table, so —
like mastering's stage/status migration — the ensure function
inspects the schema and alters when an older one is found. Same
dialect branch (SQLite =PRAGMA=, PostgreSQL =information_schema=),
portable ALTER, errors swallowed.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--migrate-entry-columns ()
  "Add timer/reschedule columns to routine_entries when missing."
  (let* ((pg (eq (and (boundp 'my/manifolding-atlas-db-backend)
                      my/manifolding-atlas-db-backend)
                 'pg))
         (info (condition-case nil
                   (if pg
                       (emacsql (manifolding-atlas-db)
                         "SELECT column_name FROM information_schema.columns WHERE table_name = 'routine_entries'")
                     (emacsql (manifolding-atlas-db)
                       "PRAGMA table_info(routine_entries)"))
                 (error nil)))
         (cols (mapcar (lambda (row)
                         (seq-find #'stringp (append row nil)))
                       info)))
    (dolist (col '(("step-duration" . "TEXT")
                   ("resched-interval" . "TEXT")
                   ("resched-days" . "TEXT")
                   ("resched-anchor" . "TEXT")
                   ("next-date" . "INTEGER")))
      (unless (member (car col) cols)
        (condition-case nil
            (emacsql (manifolding-atlas-db)
              (format "ALTER TABLE routine_entries ADD COLUMN %s %s"
                      (car col) (cdr col)))
          (error nil))))))
#+end_src

*** DB Extractor
Indexes ROUTINE_* unfoldings from NOTE-DATA. Routine notes land in
=routines=; entry headings carrying ROUTINE_ID, SOURCE and
ROUTINE_ORDER land in =routine_entries=. Session notes with SESSION_OF
and SESSION_DATE land in =routine_sessions=.

**** Routine row extraction
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--extract-routine (note-id unfoldings)
  "Index the routine row for NOTE-ID from PROPS."
  (let ((rtype (cdr (assoc "ROUTINE_TYPE" unfoldings))))
    (emacsql (manifolding-atlas-db)
      [:delete :from routines :where (= note-id $s1)]
      note-id)
    (when (and rtype
               (not (string-equal rtype "WARNING"))
               (member rtype my/manifolding-atlas-routines--types))
      (emacsql (manifolding-atlas-db)
        [:insert :into routines :values $v1]
        (vector note-id rtype
                (cdr (assoc "ROUTINE_DOMAIN" unfoldings))
                (cdr (assoc "ROUTINE_TEMPORAL" unfoldings)))))))
#+end_src
**** Entry row extraction
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--extract-entry (note-id unfoldings)
  "Index the routine_entries row for NOTE-ID from PROPS."
  (let ((rid (cdr (assoc "ROUTINE_ID" unfoldings)))
        (source (cdr (assoc "SOURCE" unfoldings)))
        (order (cdr (assoc "ROUTINE_ORDER" unfoldings))))
    (emacsql (manifolding-atlas-db)
      [:delete :from routine_entries :where (= note-id $s1)]
      note-id)
    (when (and rid source order
               (not (string-equal rid "WARNING"))
               (not (string-equal source "WARNING")))
      (emacsql (manifolding-atlas-db)
        [:insert :into routine_entries :values $v1]
        (vector note-id rid source
                (string-to-number order)
                (cdr (assoc "ROUTINE_GROUP" unfoldings))
                (cdr (assoc "STEP_DURATION" unfoldings))
                (cdr (assoc "RESCHEDULE_INTERVAL" unfoldings))
                (cdr (assoc "RESCHEDULE_DAYS" unfoldings))
                (cdr (assoc "RESCHEDULE_ANCHOR" unfoldings))
                (let ((nxt (cdr (assoc "NEXT_DATE" unfoldings))))
                  (and nxt
                       (string-match-p "^[0-9]\\{8\\}$" nxt)
                       (string-to-number nxt))))))))
#+end_src
**** Session row extraction
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--extract-session (note-id unfoldings)
  "Index the routine_sessions row for NOTE-ID from PROPS."
  (let ((sid (cdr (assoc "SESSION_OF" unfoldings)))
        (sdate (cdr (assoc "SESSION_DATE" unfoldings))))
    (emacsql (manifolding-atlas-db)
      [:delete :from routine_sessions :where (= note-id $s1)]
      note-id)
    (when (and sid sdate
               (not (string-equal sid "WARNING"))
               (not (string-equal sdate "WARNING"))
               (string-match-p "^[0-9]\\{4\\}-[0-9]\\{2\\}-[0-9]\\{2\\}$" sdate))
      (emacsql (manifolding-atlas-db)
        [:insert :into routine_sessions :values $v1]
        (vector note-id sid
                (my/manifolding-atlas-routines--datestring-int sdate)
                (string-to-number (or (cdr (assoc "SESSION_DONE" unfoldings)) "0"))
                (string-to-number (or (cdr (assoc "SESSION_SKIPPED" unfoldings)) "0"))
                (string-to-number (or (cdr (assoc "SESSION_TOTAL" unfoldings)) "0")))))))
#+end_src

*** Schema
Two schemas: =routine= requires ROUTINE_TYPE/DOMAIN/TEMPORAL on notes
tagged =routine= (but not =session=); =routines-session= requires the
SESSION_* unfoldings on notes tagged =session=.

**** Validator factories
Every field validates as required, optionally constrained to a
vocabulary — one factory per shape keeps both schemas terse.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--required-validator ()
  "Validator accepting any non-WARNING value."
  (lambda (val _n)
    (if (or (null val) (equal val "WARNING"))
        "missing-required"
      t)))

(defun my/manifolding-atlas-routines--member-validator (allowed bad-msg)
  "Validator requiring a non-WARNING member of ALLOWED."
  (lambda (val _n)
    (cond ((or (null val) (equal val "WARNING")) "missing-required")
          ((not (member val allowed)) bad-msg)
          (t t))))
#+end_src
**** Routine schema
#+begin_src emacs-lisp
(manifolding-atlas-blueprint-define
 'routine
 :predicate (lambda (note)
              (and (member "routine" (manifolding-atlas-note-tags note))

 (list
  (list :key "ROUTINE_TYPE" :type 'string
        :validate (my/manifolding-atlas-routines--member-validator

  (list :key "ROUTINE_DOMAIN" :type 'string
        :validate (my/manifolding-atlas-routines--member-validator
                   my/manifolding-atlas-routines--domains "invalid-domain"))
  (list :key "ROUTINE_TEMPORAL" :type 'string
        :validate (my/manifolding-atlas-routines--member-validator
                   (my/manifolding-atlas-routines--temporal-values)
                   "invalid-temporal"))))
#+end_src
**** Session schema
#+begin_src emacs-lisp
(manifolding-atlas-blueprint-define
 'routines-session
 :predicate (lambda (note) (member "session" (manifolding-atlas-note-tags note)))
 :include 'base
 :fields
 (list
  (list :key "SESSION_OF" :type 'string
        :validate (my/manifolding-atlas-routines--required-validator))
  (list :key "SESSION_DATE" :type 'string
        :validate (lambda (val _n)
                    (cond ((or (null val) (equal val "WARNING")) "missing-required")
                          ((not (string-match-p "^[0-9]\\{4\\}-[0-9]\\{2\\}-[0-9]\\{2\\}$" val)) "invalid-date")
                          (t t))))
  (list :key "SESSION_DONE" :type 'string
        :validate (my/manifolding-atlas-routines--required-validator))
  (list :key "SESSION_SKIPPED" :type 'string
        :validate (my/manifolding-atlas-routines--required-validator))
   (list :key "SESSION_TOTAL" :type 'string
         :validate (my/manifolding-atlas-routines--required-validator))
   (list :key "SESSION_ELAPSED" :type 'string
         :validate (lambda (val _n)
                     (cond ((or (null val) (equal val "WARNING")) t)
                           ((not (string-match-p "^[0-9]+$" val)) "invalid-elapsed")
                           (t t))))))

*** Entry Writing
Entries are appended as child headings with SOURCE, ROUTINE_ID and
ROUTINE_ORDER unfoldings. Optional group labels (Warm-Up through
Recovery) and workout prescriptions (=3x10 @20kg /180s=) are parsed
into WORKOUT_* unfoldings.

**** Prescription format
#+begin_src emacs-lisp
(defconst my/manifolding-atlas-routines--groups
  '("Warm-Up" "Main" "Accessory" "Conditioning" "Cooldown" "Recovery"))

(defconst my/manifolding-atlas-routines--rx-regexp
  "\\`\\s-*\\([0-9]+\\)\\s-*x\\s-*\\([0-9]+\\)\\(?:\\s-*@\\s-*\\([^/]+?\\)\\)?\\(?:\\s*/\\s-*\\([0-9]+\\)\\s-*s?\\)?\\s-*\\'")

**** Group and prescription prompts
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--prompt-details (items)
  "Annotate reordered ITEMS with group, prescription, duration and cadence.
ITEMS are (id . title) conses; returns
\(id title group rx duration interval days) lists. Blanks mean none;
durations look like 45s/20m, intervals like 2d/1w, days like Mon/Wed/Fri."
  (let (out)
    (dolist (item items)
      (let* ((group (completing-read
                     (format "Group for %s (blank = none): " (cdr item))
                     (append my/manifolding-atlas-routines--groups '(""))
                     nil t nil nil ""))
             (rx (read-string
                  (format "Prescription for %s (e.g. 3x10 @20kg /180, blank = none): "
                          (cdr item))))
             (duration (read-string
                        (format "Step timer for %s (e.g. 45s, blank = none): "
                                (cdr item))))
             (interval (read-string
                        (format "Repeat interval for %s (e.g. 2d/1w, blank = none): "
                                (cdr item))))
             (days (read-string
                    (format "Repeat days for %s (e.g. Mon/Wed/Fri, blank = none): "
                            (cdr item)))))
        (when (and (not (string-empty-p duration))
                   (not (atlas-timer-parse-duration duration)))
          (message "Duration %S ignored: want 45s/20m/1h30m" duration)
          (setq duration ""))
        (when (and (not (string-empty-p interval))
                   (not (my/manifolding-atlas-routines--parse-interval interval)))
          (message "Interval %S ignored: want Nd/Nw/Nm/Ny" interval)
          (setq interval ""))
        (when (and (not (string-empty-p days))
                   (not (my/manifolding-atlas-routines--parse-days days)))
          (message "Days %S ignored: want Mon/Wed/Fri" days)
          (setq days ""))
        (push (list (car item) (cdr item)
                    (unless (string-empty-p group) group)
                    (unless (string-empty-p rx) rx)
                    (unless (string-empty-p duration) duration)
                    (unless (string-empty-p interval) interval)
                    (unless (string-empty-p days) days))
              out)))
    (nreverse out)))
#+end_src
**** Prescription parsing
Unparseable prescriptions are kept verbatim in WORKOUT_RX.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--rx-parts (rx)
  "Return (sets reps load rest) parsed from RX, or nil."
  (when rx
    (when (string-match my/manifolding-atlas-routines--rx-regexp rx)
      (list (match-string 1 rx) (match-string 2 rx)
            (match-string 3 rx) (match-string 4 rx)))))
#+end_src
**** Entry heading text
Builds the full drawer for one entry heading.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--entry-text (title eid sid routine-id idx group rx duration interval days)
  "Return the entry heading text with its :PROPERTIES: drawer."
  (let* ((parts (my/manifolding-atlas-routines--rx-parts rx))
         (sets (nth 0 parts)) (reps (nth 1 parts))
         (load (nth 2 parts)) (rest (nth 3 parts)))
    (unless (or (null rx) (and sets reps))
      (message "Prescription %S stored verbatim as WORKOUT_RX" rx))
    (concat
     (format "** %s\n:PROPERTIES:\n:ID: %s\n:SOURCE: %s\n:ROUTINE_ID: %s\n:ROUTINE_ORDER: %d\n"
             title eid sid routine-id idx)
     (when group (format ":ROUTINE_GROUP: %s\n" group))
     (when sets (format ":WORKOUT_SETS: %s\n" sets))
     (when reps (format ":WORKOUT_REPS: %s\n" reps))
     (when load (format ":WORKOUT_LOAD: %s\n" load))
     (when rest (format ":WORKOUT_REST: %ss\n" rest))
     (when rx (format ":WORKOUT_RX: %s\n" rx))
     (when duration (format ":STEP_DURATION: %s\n" duration))
     (when interval (format ":RESCHEDULE_INTERVAL: %s\n" interval))
     (when days (format ":RESCHEDULE_DAYS: %s\n" days))
     ":END:\n")))
#+end_src
**** Entry writer
Dual-writes: org drawer unfoldings plus immediate routine_entries rows.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--write-entries (note items)
  "Append ITEMS as entry child headings of NOTE.
ITEMS are (id title group rx duration interval days) lists from `--prompt-details'.
Dual-writes: org drawer unfoldings plus immediate routine_entries rows."
  (let* ((file (manifolding-atlas-note-path note))
         (routine-id (manifolding-atlas-note-id note))
         (idx 0))
    (with-current-buffer (find-file-noselect file)
      (goto-char (point-max))
      (dolist (item items)
        (setq idx (1+ idx))
        (unless (bolp) (insert "\n"))
        (let ((eid (org-id-new)))
          (insert (my/manifolding-atlas-routines--entry-text
                   (nth 1 item) eid (nth 0 item) routine-id idx
                   (nth 2 item) (nth 3 item)
                   (nth 4 item) (nth 5 item) (nth 6 item)))
          (org-id-add-location eid file)
          (my/manifolding-atlas-routines--entry-insert-row
           eid routine-id (nth 0 item) idx (nth 2 item)
           (nth 4 item) (nth 5 item) (nth 6 item))))
      (save-buffer))))
#+end_src

*** Queries
Entries are read from the DB index first (fast), falling back to a
direct org scan when unsynced. Workout detail always reads live from
the routine file.

**** Browsing

  (let ((ids (mapcar #'car

**** Routine selection helpers
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--routine-candidates ()
  "Return alist of (label . note-id) for every known routine."
  (mapcar (lambda (row)
            (cons (format "%s (%s)" (cadr row) (car row)) (car row)))
          (emacsql (manifolding-atlas-db)
                   [:select [routines:note-id notes:title]
                    :from routines
                    :inner :join notes
                    :on (= routines:note-id notes:id)])))

(defun my/manifolding-atlas-routines--select-routine (prompt candidates)
  "Completing-read a routine from PROMPT and CANDIDATES, return its id."
  (cdr (assoc (completing-read prompt (mapcar #'car candidates) nil nil)
              candidates)))

(defun my/manifolding-atlas-routines--select-routine-interactive (prompt)
  "Prompt for a routine and return the note."
  (let ((candidates (my/manifolding-atlas-routines--routine-candidates)))
    (when (null candidates) (user-error "No routines defined yet"))
    (manifolding-atlas-db-get-by-id
     (my/manifolding-atlas-routines--select-routine prompt candidates))))
#+end_src
**** Org fallback reader
Used when the DB index has no rows for a routine.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--entries-org (routine)
  "Read entries of ROUTINE directly from org as fallback.
Returns (source-id title order group sets reps load rest rx duration
interval days anchor next) lists."
  (let ((entries nil))
    (with-current-buffer
        (find-file-noselect (manifolding-atlas-note-path routine))
      (org-map-entries
       (lambda ()
         (let ((source (org-entry-get nil "SOURCE"))
               (order  (org-entry-get nil "ROUTINE_ORDER"))
               (group  (org-entry-get nil "ROUTINE_GROUP")))
           (when (and source order)
             (push (list source
                         (org-get-heading t t t t)
                         (string-to-number order)
                         group
                         (org-entry-get nil "WORKOUT_SETS")
                         (org-entry-get nil "WORKOUT_REPS")
                         (org-entry-get nil "WORKOUT_LOAD")
                         (org-entry-get nil "WORKOUT_REST")
                         (org-entry-get nil "WORKOUT_RX")
                         (org-entry-get nil "STEP_DURATION")
                         (org-entry-get nil "RESCHEDULE_INTERVAL")
                         (org-entry-get nil "RESCHEDULE_DAYS")
                         (org-entry-get nil "RESCHEDULE_ANCHOR")
                         (org-entry-get nil "NEXT_DATE"))
                   entries))))
       nil nil))
    (sort (nreverse entries)
          (lambda (a b) (< (nth 2 a) (nth 2 b))))))
#+end_src
**** Live workout detail
Workout unfoldings always come from the file, never the stale index.

(defun my/manifolding-atlas-routines--session-create (routine results done skipped elapsed)
  "Create or update today's session note for ROUTINE.
RESULTS is a list of (source-id title result elapsed planned grade),
result being `done or `skip. ELAPSED is whole-run wall seconds."

                 date (manifolding-atlas-title-to-slug rtitle)))
          (unfoldings `(("SESSION_OF" . ,rid)
                   ("SESSION_DATE" . ,date)

*** Streaks
Temporal-aware streak counting: the period unit derives from the
routine's TEMPORAL unfolding. A streak survives if the current or
previous period has a session.

**** Period mapping and bucketing
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--streak-unit (temporal)
  "Map TEMPORAL context to a streak period unit."
  (pcase temporal
    ("WEEKLY" 'week)
    ("MONTHLY" 'month)
    ((or "QUARTERLY" "SEASONAL") 'quarter)
    ("YEARLY" 'year)
    (_ 'day)))

(defun my/manifolding-atlas-routines--bucket (date-int unit)
  "Bucket YYYYMMDD DATE-INT into an ordinal of UNIT."
  (let* ((y (/ date-int 10000))
         (m (% (/ date-int 100) 100))
         (d (% date-int 100))
         (days (time-to-days (encode-time 0 0 0 d m y))))
    (pcase unit
      ('week (/ days 7))
      ('month (+ (* y 12) m))
      ('quarter (+ (* y 4) (1+ (/ (1- m) 3))))
      ('year y)
      (_ days))))
#+end_src
**** Streak counting
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--streak-for (routine-id &optional temporal)
  "Consecutive-period streak of ROUTINE-ID ending this or the previous
period. The period derives from TEMPORAL (day by default; week, month,
quarter/season, or year for the matching temporal contexts)."
  (let* ((unit (my/manifolding-atlas-routines--streak-unit temporal))
         (buckets (delete-dups
                   (sort (mapcar (lambda (row)
                                   (my/manifolding-atlas-routines--bucket
                                    (nth 0 row) unit))
                                 (my/manifolding-atlas-routines--sessions-for
                                  routine-id))
                         #'>)))
         (today-int (my/manifolding-atlas-routines--datestring-int
                     (my/manifolding-atlas-routines--today-string)))
         (anchor (my/manifolding-atlas-routines--bucket today-int unit))
         (streak 0))
    (when buckets
      (cond ((= (car buckets) anchor))
            ((= (car buckets) (1- anchor)) (setq anchor (car buckets)))
            (t (setq buckets nil)))
      (while (and buckets (= (car buckets) anchor))
        (setq streak (1+ streak)
              anchor (1- anchor)
              buckets (cdr buckets))))
    streak))

*** Anchored Rescheduling
Elisp-native port of the reschedule-by-rule idea: a stable anchor
(last completion date) plus an interval and/or weekday set decide the
next due date, immune to manual drag-drift. No Python, no croniter.
Computed at session completion for done entries; skipped entries
never advance. Each engine owns its copy of this logic — routines and
protocols intentionally do not share it.

**** Interval and weekday parsing
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--parse-interval (str)
  "Parse STR like \"2d\"/\"1w\"/\"3m\"/\"1y\" into (NUM . UNIT).
UNIT is one of d/w/m/y. Return nil when STR is not an interval.
\"h\" parses as (NUM . h) but anchor math is day-precision, so hour
intervals resolve to the same day; intra-day repetition belongs to
the run, not the anchor."
  (when (and (stringp str)
             (string-match "\\`\\s-*\\([0-9]+\\)\\s-*\\([hdwmy]\\)\\s-*\\'" str))
    (cons (string-to-number (match-string 1 str))
          (intern (match-string 2 str)))))

(defconst my/manifolding-atlas-routines--dow-names
  ["Sun" "Mon" "Tue" "Wed" "Thu" "Fri" "Sat"]
  "Three-letter weekday names indexed like `decoded-time-weekday'.")

(defun my/manifolding-atlas-routines--parse-days (str)
  "Parse STR like \"Mon/Wed/Fri\" into a sorted list of weekday numbers.
Sunday is 0. Accept full names, any case, separated by slash, comma
or whitespace. Return nil when nothing parses."
  (when (stringp str)
    (let (out)
      (dolist (tok (split-string str "[/, \t]+" t))
        (let ((pos nil)
              (low (downcase tok)))
          (cl-loop for i from 0 to 6
                   for full = (aref my/manifolding-atlas-routines--dow-names i)
                   for flow = (downcase full)
                   when (or (string= low flow)
                            (and (>= (length low) 2)
                                 (or (string-prefix-p low flow)
                                     (string-prefix-p flow low))))
                   do (setq pos i))
          (when pos (push pos out))))
      (sort (delete-dups out) #'<))))
#+end_src

**** Date arithmetic
Day-precision ISO helpers. Months clamp (Jan 31 + 1m = Feb 28/29),

#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--iso-days (iso)
  "Convert ISO date \"YYYY-MM-DD\" to an org day-number."
  (time-to-days
   (encode-time 0 0 0
                (string-to-number (substring iso 8 10))
                (string-to-number (substring iso 5 7))
                (string-to-number (substring iso 0 4)))))

(defun my/manifolding-atlas-routines--days-iso (days)
  "Convert an org day-number back to ISO \"YYYY-MM-DD\".
Note `days-to-time' uses the Emacs epoch, not `time-to-days'' —
the offset must be subtracted first."
  (format-time-string "%Y-%m-%d"
                      (days-to-time (- days (time-to-days 0)))))

(defun my/manifolding-atlas-routines--iso-weekday (iso)
  "Weekday number of ISO date: Sunday 0 through Saturday 6."
  (nth 6 (decode-time (encode-time 0 0 0
                                   (string-to-number (substring iso 8 10))
                                   (string-to-number (substring iso 5 7))
                                   (string-to-number (substring iso 0 4))))))

(defun my/manifolding-atlas-routines--iso-add (iso num unit)
  "Add NUM UNITs (d/w/m/y/h) to ISO date; return ISO.
\"h\" is sub-day and returns ISO unchanged (day precision)."
  (require 'calendar)
  (pcase unit
    ('h iso)
    ('d (my/manifolding-atlas-routines--days-iso
         (+ (my/manifolding-atlas-routines--iso-days iso) num)))
    ('w (my/manifolding-atlas-routines--days-iso
         (+ (my/manifolding-atlas-routines--iso-days iso) (* 7 num))))
    ((or 'm 'y)
     (let* ((y (string-to-number (substring iso 0 4)))
            (m (string-to-number (substring iso 5 7)))
            (d (string-to-number (substring iso 8 10)))
            (total (+ (1- m) (if (eq unit 'm) num (* 12 num))))
            (ny (+ y (/ total 12)))
            (nm (1+ (% total 12)))
            (last (calendar-last-day-of-month nm ny)))
       (format "%04d-%02d-%02d" ny nm (min d last))))))

**** Next-date computation
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--next-date (anchor today interval days)
  "Next due ISO date strictly after TODAY.
ANCHOR is the last-completion ISO date (or nil for first run).
INTERVAL is a (NUM . UNIT) from `--parse-interval' (or nil).
DAYS is a weekday-number list from `--parse-days' (or nil).
Lattice rule: step from the anchor by INTERVAL until past TODAY;
with DAYS, keep stepping until a listed weekday lands. DAYS alone
means the next listed weekday after TODAY. Neither means nil."
  (when (or interval days)
    (let ((base (or anchor today))
          (today-n (my/manifolding-atlas-routines--iso-days today))
          (cand nil)
          (i 0))
      (if interval
          (progn
            (setq cand base)
            (while (and (<= (my/manifolding-atlas-routines--iso-days cand)
                            today-n)
                        (< (cl-incf i) 1000))
              (setq cand (my/manifolding-atlas-routines--iso-add
                          cand (car interval) (cdr interval))))
            (when days
              (setq i 0)
              (while (and (not (member (my/manifolding-atlas-routines--iso-weekday cand)
                                       days))
                          (< (cl-incf i) 1000))
                (setq cand (my/manifolding-atlas-routines--iso-add
                            cand (car interval) (cdr interval))))))
        (setq cand (my/manifolding-atlas-routines--days-iso (1+ today-n))
              i 0)
        (while (and (not (member (my/manifolding-atlas-routines--iso-weekday cand)
                                 days))
                    (< (cl-incf i) 8))
          (setq cand (my/manifolding-atlas-routines--days-iso
                      (1+ (my/manifolding-atlas-routines--iso-days cand))))))
      (when (< i 1000) cand))))
#+end_src

**** Advance writer
Sole writer of entry next-dates. Writes three facets at once: the
RESCHEDULE_ANCHOR and NEXT_DATE drawer unfoldings (machine truth the
extractor indexes) and the SCHEDULED planning line (the org/agenda
facet). No org-schedule call, no hooks — a plain planning-line edit,
so org can only ever display these dates, never decide them.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--write-entry-next (routine sid anchor next)
  "Stamp ANCHOR (ISO) and NEXT (ISO) onto ROUTINE's entries for SID.
Writes RESCHEDULE_ANCHOR + NEXT_DATE unfoldings and the SCHEDULED
planning line; updates the routine_entries row to match."
  (let ((file (manifolding-atlas-note-path routine))
        (next-int (string-to-number (concat (substring next 0 4)
                                            (substring next 5 7)
                                            (substring next 8 10))))
        (sched (format "<%s %s>" next
                       (aref my/manifolding-atlas-routines--dow-names
                             (my/manifolding-atlas-routines--iso-weekday next)))))
    (with-current-buffer (find-file-noselect file)
      (org-map-entries
       (lambda ()
         (when (string= (org-entry-get nil "SOURCE") sid)
           (org-entry-put nil "RESCHEDULE_ANCHOR" anchor)
           (org-entry-put nil "NEXT_DATE"
                          (format "%d" next-int))
           (save-excursion
             (org-back-to-heading t)
             (let ((end (save-excursion (org-end-of-subtree t t))))
               (forward-line 1)
               (if (re-search-forward "^SCHEDULED:" end t)
                   (progn (beginning-of-line)
                          (delete-region (line-beginning-position)
                                         (line-end-position))
                          (insert (format "SCHEDULED: %s" sched)))
                 (org-back-to-heading t)
                 (end-of-line)
                 (insert (format "\nSCHEDULED: %s" sched)))))))
       nil nil)
      (save-buffer))
    (condition-case nil
        (emacsql (manifolding-atlas-db)
          [:update routine_entries
           :set resched-anchor = $s1, next-date = $s2
           :where (and (= routine-id $s3) (= source-id $s4))]
          anchor next-int
          (manifolding-atlas-note-id routine) sid)
      (error nil))))

(defun my/manifolding-atlas-routines--advance-anchors (routine results)
  "Advance rescheduling for done entries in RESULTS.
RESULTS are (source-id title result ...) lists. Done entries step
their anchor to today and compute the next date; skipped entries
and entries without cadence are left alone."
  (let ((today (my/manifolding-atlas-routines--today-string))
        (advanced 0))
    (dolist (res results)
      (when (eq (nth 2 res) 'done)
        (let* ((sid (nth 0 res))
               (cad (my/manifolding-atlas-routines--entry-cadence
                     routine sid)))
          (when cad
            (let ((next (my/manifolding-atlas-routines--next-date
                         (nth 2 cad) today (nth 0 cad) (nth 1 cad))))
              (when next
                (my/manifolding-atlas-routines--write-entry-next
                 routine sid today next)
                (cl-incf advanced)))))))
    (when (> advanced 0)
      (message "Rescheduled %d entries" advanced))))

(defun my/manifolding-atlas-routines--entry-cadence (routine sid)
  "Return (INTERVAL DAYS ANCHOR) for ROUTINE's first entry with SID.
INTERVAL/DAYS are parsed (nil when absent); ANCHOR is the raw ISO
string (nil when absent). Return nil when the entry has no cadence."
  (let (found)
    (with-current-buffer
        (find-file-noselect (manifolding-atlas-note-path routine))
      (org-map-entries
       (lambda ()
         (when (and (not found)
                    (string= (org-entry-get nil "SOURCE") sid))
           (let ((iv (my/manifolding-atlas-routines--parse-interval
                      (org-entry-get nil "RESCHEDULE_INTERVAL")))
                 (dy (my/manifolding-atlas-routines--parse-days
                      (org-entry-get nil "RESCHEDULE_DAYS"))))
             (when (or iv dy)
               (setq found
                     (list iv dy
                           (org-entry-get nil "RESCHEDULE_ANCHOR")))))))
       nil nil))
    found))

(defun my/manifolding-atlas-routines--run-step (buf entry step total)
  "Interactively run ENTRY as step STEP of TOTAL in review BUF."
  (let* ((sid (nth 0 entry))
         (stitle (nth 1 entry))
         (snote (manifolding-atlas-db-get-by-id sid))
         (mastery (my/manifolding-atlas-routines--mastery-status sid))
         (badge (car mastery))
         (due-p (cdr mastery))
         (planned (atlas-timer-parse-duration (nth 9 entry)))
         (t0 (atlas-timer-stopwatch-start))
         (base-choices (if due-p '(?l ?r ?d ?s ?o ?q) '(?d ?s ?o ?q)))
         (choices (append base-choices
                          (when planned '(?t ?+ ?-))))

          (concat (format "[%d/%d] %s%s — "
                          step total stitle

                  (if due-p
                      "l=learn+done r=review-only d=done s=skip o=open q=quit: "
                    "d=done s=skip o=open q=quit: "))))

      (atlas-timer-countdown-start planned)
      (setq prompt (concat prompt "t=pause +=+1m -=-1m: ")))
    (my/manifolding-atlas-routines--display-step
     buf step total stitle sid badge planned
     (and planned (atlas-timer-countdown-remaining)))
    (let ((done nil)
          (answer nil))
      (while (not done)
        (setq answer (read-char-choice prompt choices))
        (pcase answer
          (?t (condition-case nil
                  (atlas-timer-countdown-toggle-pause)
                (error (message "No active countdown")))
              (my/manifolding-atlas-routines--display-step
               buf step total stitle sid badge planned
               (atlas-timer-countdown-remaining)))
          (?+ (setq planned (+ (or planned 0) 60))
              (atlas-timer-countdown-start
               (+ (or (atlas-timer-countdown-remaining) 0) 60))
              (my/manifolding-atlas-routines--display-step
               buf step total stitle sid badge planned
               (atlas-timer-countdown-remaining)))
          (?- (setq planned (max 0 (- (or planned 0) 60)))
              (atlas-timer-countdown-start
               (max 0 (- (or (atlas-timer-countdown-remaining) 0) 60)))
              (my/manifolding-atlas-routines--display-step
               buf step total stitle sid badge planned
               (atlas-timer-countdown-remaining)))
          (_ (setq done t))))
      (let ((elapsed (atlas-timer-stopwatch-elapsed t0))
            (grade nil)
            (result nil))
        (atlas-timer-countdown-stop)
        (setq result
              (pcase answer
                (?l (setq grade (my/manifolding-atlas-routines--grade-step sid))
                    '(done . t))
                (?r (setq grade (my/manifolding-atlas-routines--grade-step sid))
                    (message "Reviewed without marking practice.") nil)
                (?d '(done . nil))
                (?s '(skip . nil))
                (?o (let ((res (my/manifolding-atlas-routines--open-followup
                                snote sid due-p)))
                      (cond ((eq res 'quit) 'quit)
                            ((eq res 'done-learn)
                             (setq grade (my/manifolding-atlas-routines--grade-step sid))
                             '(done . t))
                            (res (cons res nil))
                            (t nil))))
                (?q 'quit)))
        (push (list sid elapsed planned grade)
              my/manifolding-atlas-routines--step-timings)
        result))))
#+end_src

(defun my/manifolding-atlas-routines--attach-timing (sid title result)
  "Build a result tuple (SID TITLE RESULT ELAPSED PLANNED GRADE).
Timing and learn-grade come from `--step-timings'; missing entries
mean untimed and ungraded."
  (let ((tim (assoc sid my/manifolding-atlas-routines--step-timings)))
    (list sid title result
          (and tim (nth 1 tim))
          (and tim (nth 2 tim))
          (and tim (nth 3 tim)))))

(defun my/manifolding-atlas-routines-run (&optional routine)
  "Execute ROUTINE (or prompt for one) step by step in a review buffer.
Steps that are in the mastering pool show their review status; due
steps can be graded with r (learn) before being practiced. Steps
with STEP_DURATION run a countdown (t pauses, +/- nudge a minute);
the whole run is stopwatched and per-step times land in the session
journal. Done entries advance their reschedule anchors. At the
end, offer to journal the session."
  (interactive)
  (let* ((routine (or routine
                      (my/manifolding-atlas-routines--select-routine-interactive
                       "Run routine: ")))
         (entries (my/manifolding-atlas-routines--entries routine)))
    (unless entries (user-error "Routine has no entries"))
    (let ((buf (get-buffer-create "*manifolding-atlas-routines-run*"))
          (total (length entries))
          (done 0)
          (skipped 0)
          (learned 0)
          (step 0)
          (results nil)
          (my/manifolding-atlas-routines--step-timings nil)
          (t0 (atlas-timer-stopwatch-start)))
      (with-current-buffer buf
        (special-mode)
        (setq buffer-read-only t))
      (pop-to-buffer-same-window buf)
      (catch 'routines-quit
        (dolist (entry entries)
          (setq step (1+ step))
          (let ((res (my/manifolding-atlas-routines--run-step
                      buf entry step total)))
            (cond
             ((eq res 'quit) (throw 'routines-quit nil))
             (res
              (pcase (car res)
                ('done (setq done (1+ done)))
                ('skip (setq skipped (1+ skipped))))
              (when (cdr res) (setq learned (1+ learned)))
              (push (my/manifolding-atlas-routines--attach-timing
                     (nth 0 entry) (nth 1 entry) (car res))
                    results))))))
      (kill-buffer buf)
      (atlas-timer-countdown-stop)
      (let ((elapsed (atlas-timer-stopwatch-elapsed t0)))
        (when results
          (my/manifolding-atlas-routines--advance-anchors
           routine (nreverse results)))
        (when (and results
                   (y-or-n-p (format "Log session (%d done / %d skipped / %d learned)? "
                                     done skipped learned)))
          (my/manifolding-atlas-routines--session-create
           routine (nreverse results) done skipped elapsed))
        (message "Routine finished: %d done, %d skipped, %d learned in %s"
                 done skipped learned
                 (atlas-timer-format-seconds elapsed))))))
#+end_src

(defun my/manifolding-atlas-routines--add-entry-source ()
  "Return (id . title) for a new entry source."
  (if (and (derived-mode-p 'org-mode)
           (org-at-heading-p)
           (buffer-file-name)
           (file-in-directory-p
            (buffer-file-name)
            (my/manifolding-atlas-root-dir)))
      (progn (org-id-get-create)
             (when buffer-file-name (save-buffer))
             (cons (org-id-get) (org-get-heading t t t t)))
    (let* ((notes (manifolding-atlas-db-query))
           (cands (mapcar (lambda (n)
                            (cons (manifolding-atlas-note-title n)
                                  (manifolding-atlas-note-id n)))
                          notes))
           (choice (completing-read "Entry note: "
                                    (mapcar #'car cands) nil t)))
      (cons (cdr (assoc choice cands)) choice))))

**** Order correction
write-entries assigns sequential orders; a manual append forces the
next free order instead.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--fix-entry-order (routine next-order)
  "Rewrite the last entry's ROUTINE_ORDER on ROUTINE to NEXT-ORDER."
  (with-current-buffer (find-file-noselect
                        (manifolding-atlas-note-path routine))
    (goto-char (point-max))
    (when (re-search-backward ":ROUTINE_ORDER: [0-9]+" nil t)
      (let ((inhibit-read-only t))
        (delete-region (match-beginning 0) (match-end 0))
        (insert (format ":ROUTINE_ORDER: %d" next-order))))
    (save-buffer)))

(defun my/manifolding-atlas-routines-add-entry (&optional routine)
  "Append an entry to ROUTINE (or prompt for one).
Source is the heading at point when in a vault note, else a
vault-wide selection.  Regenerates the rendered file afterwards."
  (interactive)
  (let* ((routine (or routine
                      (my/manifolding-atlas-routines--select-routine-interactive

         (sid (my/manifolding-atlas-routines--add-entry-source))
         (existing (my/manifolding-atlas-routines--entries-org routine))

     routine (list (list (car sid) (cdr sid) nil nil)))
    (my/manifolding-atlas-routines--fix-entry-order routine next-order)
    (let ((note routine))
      (my/manifolding-atlas-routines-generate note)
      (message "Entry added to %s" (manifolding-atlas-note-title routine)))
    routine))

(defun my/manifolding-atlas-routines--clear-entry-headings (note)
  "Delete all child entry headings of NOTE."
  (with-current-buffer (find-file-noselect
                        (manifolding-atlas-note-path note))
    (goto-char (point-min))
    (while (re-search-forward "^\\*\\* " nil t)
      (org-back-to-heading t)
      (let ((beg (point)))
        (org-end-of-subtree t t)
        (delete-region beg (point))))
    (save-buffer)))

(defun my/manifolding-atlas-routines-due-reviews ()
  "Create or refresh the \"Due Reviews\" composite routine from
today's due mastering notes.  Stable ID across refreshes; renders
like any other routine and appears in streaks/report."
  (interactive)
  (let ((ids (my/manifolding-atlas-mastering--due-note-ids)))
    (when (null ids) (user-error "Nothing due today"))
    (let* ((title "Due Reviews")
           ;; Reuse an existing Due Reviews routine if present.
           (rows (emacsql (manifolding-atlas-db)
                          [:select [routines:note-id notes:title]
                           :from routines
                           :inner :join notes :on (= routines:note-id notes:id)]))
           (rid (cl-loop for row in rows
                         when (string-equal (cadr row) title)
                         return (car row)))
           (note (if rid
                     (manifolding-atlas-db-get-by-id rid)
                   (my/manifolding-atlas--create-file-wrapper
                    title
                    (concat "routines/due-reviews")
                    '("routine")
                    '(("ROUTINE_TYPE" . "composite"))
                    "")))
           (items (mapcar (lambda (id)
                            (let ((n (manifolding-atlas-db-get-by-id id)))
                              (cons id (if n (manifolding-atlas-note-title n)
                                         id))))
                          ids))
           (ordered (my/manifolding-atlas-routines--reorder items))
           (details (my/manifolding-atlas-routines--prompt-details ordered)))
      (unless details (user-error "Aborted"))
      (my/manifolding-atlas-routines--clear-entry-headings note)
      (my/manifolding-atlas-routines--write-entries note details)
      (my/manifolding-atlas-routines-generate note)
      (message "Due Reviews refreshed (%d entries)" (length details))
      note)))
