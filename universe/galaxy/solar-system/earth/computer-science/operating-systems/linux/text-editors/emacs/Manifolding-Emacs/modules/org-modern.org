  (setq org-modern-hide-stars "  "
        org-modern-fold-stars
        '(("◉" . "◯")
          ("│" . "└")
          (" │" . " └")
          (" │" . " └"))
        org-modern-keyword
        '(("title"      . "Title")
          ("filetags"   . "Tags")
          ("identifier" . "ID")
          ("date"       . "Date")
          ("author"     . "Author")
          ("priority"   . "Priority"))
        org-modern-property-drawer t
        org-modern-links t
        )

  (unless (display-graphic-p)
    (setq org-modern-block-fringe    nil
          org-modern-table-vertical  1
          org-modern-table-horizontal 0
          org-modern-star            nil))

  (advice-add 'set-face-attribute :around
              (lambda (orig face frame &rest args)
                (condition-case nil
                    (apply orig face frame args)
                  (error nil))))

  (global-org-modern-mode)
