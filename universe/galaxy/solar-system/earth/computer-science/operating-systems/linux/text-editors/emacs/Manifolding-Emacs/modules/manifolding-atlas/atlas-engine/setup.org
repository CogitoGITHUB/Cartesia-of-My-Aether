** DB relocation guard
Guarantees the vault DB always lives beside the notes even if a
stale session opened it elsewhere.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-db--relocate-legacy ()
  "Move a default-location DB to `manifolding-atlas-db-location'.
Runs before every connection attempt; cheap no-op once relocated."
  (let ((legacy (expand-file-name "manifolding-atlas.db"
                                  user-emacs-directory))
        (target manifolding-atlas-db-location))
    (when (and (file-exists-p legacy)
               (not (file-exists-p target))
               (not (equal (expand-file-name legacy)
                           (expand-file-name target))))
      (make-directory (file-name-directory target) t)
      (condition-case err
          (progn
            (manifolding-atlas-db-close)
            (rename-file legacy target nil)
            (message "Manifolding Atlas: relocated database to %s"
                     target))
        (error
         (lwarn 'manifolding-atlas :warning
                "database relocation failed: %s"
                (error-message-string err)))))))

(defun my/manifolding-atlas-db-location-guard (orig)
  "Reconnect when the DB endpoint changed under us.
Tracks backend + location as one endpoint identity: sqlite reacts to
path changes (plus the legacy file relocation), pg reacts to backend
flips and ignores path changes (its data lives outside the vault)."
  (unless my/manifolding-atlas-db--guard-inside
    (let ((my/manifolding-atlas-db--guard-inside t))
      (when (eq my/manifolding-atlas-db-backend 'sqlite)
        (my/manifolding-atlas-db--relocate-legacy))
      (let ((endpoint (cons my/manifolding-atlas-db-backend
                            (and (eq my/manifolding-atlas-db-backend 'sqlite)
                                 manifolding-atlas-db-location))))
        (when (and manifolding-atlas-db--connection
                   (not (equal my/manifolding-atlas-db--location-used endpoint)))
          (manifolding-atlas-db-close))
        (when (or (null manifolding-atlas-db--connection)
                  (not (equal my/manifolding-atlas-db--location-used endpoint)))
          (funcall orig)
          (setq my/manifolding-atlas-db--location-used endpoint)))))
  (funcall orig))

(advice-add 'manifolding-atlas-db :around
            #'my/manifolding-atlas-db-location-guard)
#+end_src

** Sync overrides: fswatch sentinel and file predicate
Tracks `.org' files AND extensionless vault files.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-db-sync--fswatch-sentinel (proc event)
  (unless (or (string-match-p "hangup\\|finished" event)
              (process-live-p proc))
    (message "Manifolding Atlas: fswatch process died (%s), restarting..."
             (string-trim event))))

(advice-add 'manifolding-atlas-db-sync--fswatch-sentinel :override
            #'my/manifolding-atlas-db-sync--fswatch-sentinel)

(defun my/manifolding-atlas-db-sync--org-file-p (path)
  "Track `.org' files AND extensionless vault files."
  (and path
       (not (string-match-p "/\\." path))
       (or (seq-some (lambda (ext) (string-suffix-p ext path))
                     (manifolding-atlas-db--all-extensions))
           (not (string-match-p "\\." (file-name-nondirectory path))))))

(advice-add 'manifolding-atlas-db-sync--org-file-p :override
            #'my/manifolding-atlas-db-sync--org-file-p)
#+end_src

** Sync override: race-safe directory walk
Walks manually instead of `directory-files-recursively' so that:
`.git' is never descended into (it is not content), and a directory
that vanishes mid-scan (git gc/pack races) is skipped instead of
killing the whole poll cycle.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-db-sync--list-org-files (dir)
  "List vault files under DIR: `.org' plus extensionless entries."
  (let* ((root (expand-file-name dir))
         (files nil)
         (dirs (list root))
         (seen (make-hash-table :test #'equal)))
    (while dirs
      (let ((d (pop dirs)))
        (when (and (file-directory-p d)
                   (not (gethash d seen)))
          (puthash d t seen)
          ;; "^[^.]" skips dotfiles/dotdirs (.git included) at the source.
          (let ((listing (ignore-errors (directory-files d t "^[^.]" t))))
            (dolist (entry listing)
              (cond
               ((file-directory-p entry)
                (unless (string= (file-name-nondirectory entry) ".git")
                  (push entry dirs)))
               ((file-regular-p entry)
                (push entry files))))))))
    (seq-filter #'manifolding-atlas-db-sync--org-file-p
                (nreverse files))))

(advice-add 'manifolding-atlas-db-sync--list-org-files :override
            #'my/manifolding-atlas-db-sync--list-org-files)

(defun my/manifolding-atlas-db-sync--scan-files-async (dirs callback)
  "List tracked files in DIRS (extensionless included), call CALLBACK."
  (let ((files (mapcan (lambda (d)
                         (my/manifolding-atlas-db-sync--list-org-files d))
                       dirs)))
    (run-at-time 0 nil callback files)))

(advice-add 'manifolding-atlas-db-sync--scan-files-async :override
            #'my/manifolding-atlas-db-sync--scan-files-async)
#+end_src

** Sync override: parse method
Manifolding Atlas forces non-'.org' files (extensionless vault notes) through
`find-file', which trips over atlas-wormholes's read-only buffers.
Always use the configured method instead.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-db--parse-file (path)
  "Parse PATH honoring `manifolding-atlas-db-parse-method'."
  (let* ((manifolding-atlas-db--active-parse-method
          manifolding-atlas-db-parse-method)
         (method manifolding-atlas-db-parse-method))
    (pcase method
      ('single-temp-buffer
       (manifolding-atlas-db--parse-with-temp-buffer path nil))

      ('temp-buffer
       (manifolding-atlas-db--parse-with-temp-buffer path t))

      (_
       (let ((buffer (find-file-noselect path)))
         (unwind-protect
             (with-current-buffer buffer
               (let* ((ast
                       (org-element-parse-buffer
                        (manifolding-atlas-db--effective-granularity)))
                      (attrs (file-attributes path))
                      (mtime
                       (float-time
                        (file-attribute-modification-time attrs)))
                      (size (file-attribute-size attrs))
                      (hash (secure-hash 'sha256 (current-buffer)))
                      (file-title
                       (manifolding-atlas-db--extract-file-title ast path)))
                 (make-manifolding-atlas-parse-ctx
                  :path path
                  :ast ast
                  :file-node
                  (manifolding-atlas-db--extract-file-node
                   ast path (current-buffer) file-title)
                  :heading-nodes
                  (manifolding-atlas-db--extract-heading-nodes
                   ast path (current-buffer) file-title)
                  :hash hash
                  :mtime mtime
                  :size size)))

           (when (buffer-live-p buffer)
             (kill-buffer buffer))))))))

(advice-add 'manifolding-atlas-db--parse-file :override
            #'my/manifolding-atlas-db--parse-file)
#+end_src

** Deferred database open
Ensure the database exists shortly after boot, but late enough that
the initial full-vault index never competes with startup. The first
connection parses the whole vault and can block for a while.
#+begin_src emacs-lisp
(with-eval-after-load 'manifolding-atlas
  (run-with-idle-timer
   15 nil
   (lambda ()
     (message "Manifolding Atlas: opening database (may pause briefly)...")
     (condition-case err
         (progn
           (manifolding-atlas-db)
           (manifolding-atlas-db-autosync-mode +1)
           (message "Manifolding Atlas: database ready (%s)"
                    (file-name-nondirectory
                     manifolding-atlas-db-location)))
       (error (lwarn 'manifolding-atlas :error
                     "database init failed: %s"
                     (error-message-string err)))))))
#+end_src
