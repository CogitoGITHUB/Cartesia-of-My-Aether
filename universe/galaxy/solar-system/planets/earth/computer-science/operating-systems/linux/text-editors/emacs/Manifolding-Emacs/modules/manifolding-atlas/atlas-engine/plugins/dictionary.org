* Database Table
#+begin_src emacs-lisp
(defun my/manifolding-atlas-dictionary--ensure-table ()
  "Create the dictionary_words table if it does not exist."
  (emacsql (manifolding-atlas-db)
    [:create :table :if :not :exists dictionary_words
     ([(note-id :not-null)
       (word :not-null)
       (pos :null)
       (source :null)
       (added :null)]
      (:primary-key [note-id])
      (:foreign-key [note-id] :references notes [id]
       :on-delete :cascade))]))
#+end_src

* DB Extractor
Reads WORD/POS/SOURCE/ADDED straight from note-data's unfoldings and
title — no AST walk needed, so this stays fast and worker-eligible.

** Extractor
#+begin_src emacs-lisp
(defun my/manifolding-atlas-dictionary--extract (_ctx note-data)
  "Extract dictionary fields from NOTE-DATA and store in dictionary_words table."
  (let* ((note-id (plist-get note-data :id))
         (tags    (plist-get note-data :tags))
         (unfoldings   (plist-get note-data :unfoldings)))
    (when (member "dictionary" tags)
      (let ((word   (or (cdr (assoc "WORD" unfoldings))
                        (plist-get note-data :title)))
            (pos    (cdr (assoc "POS" unfoldings)))
            (source (cdr (assoc "SOURCE" unfoldings)))
            (added  (cdr (assoc "ADDED" unfoldings))))
        (emacsql (manifolding-atlas-db)
          [:delete :from dictionary_words :where (= note-id $s1)]
          note-id)
        (emacsql (manifolding-atlas-db)
          [:insert :into dictionary_words :values $v1]
          (vector note-id word pos source added))))
    note-data))
#+end_src

          :required t

          :validate (lambda (val _note)

          :type 'string
          :required t
          :validate (lambda (val _note)
                      (if (equal val "WARNING") "missing-required" t)))

* Word Capture
Dedicated capture command rather than the global prompt registry —
POS/SOURCE prompts should only ever fire for dictionary notes, not every
file-level note in the vault.

** Slugify helper
#+begin_src emacs-lisp
(defun my/manifolding-atlas-dictionary--slugify (word)
  "Turn WORD into a safe file-name stem."
  (let ((s (downcase word)))
    (setq s (replace-regexp-in-string "[^a-z0-9]+" "-" s))
    (replace-regexp-in-string "^-+\\|-+$" "" s)))
#+end_src

(defun my/manifolding-atlas-dictionary--open-body-capture (note body)
  "Open org-capture for NOTE's body with BODY template."
  (let ((org-capture-templates
         `(("v" "Edit note body"
             plain
             (file+function
              ,(manifolding-atlas-note-path note)
              (lambda ()
                (goto-char (point-max))
                (skip-chars-backward " \t\n")))
             ,(or body "%?")
             :kill-buffer t :unnarrowed t))))
    (org-capture nil "v"))
  (set-buffer-modified-p nil)
  (when (fboundp 'modaled-set-state) (modaled-set-state "insert")))

** New word command
Creates a new dictionary note for WORD. Tags are set automatically:
`dictionary', the slugified WORD, and each subdirectory level under
`dictionary/'.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-dictionary-new-word (word)
  "Create a new dictionary note for WORD.
Prompts for subdirectory under `dictionary/', fast mode level,
dictionary-specific prompts (POS, pronunciation, etc.), and
general prompts (state, tags, etc.)."
  (interactive "sWord: ")
  (let* ((root (expand-file-name "dictionary/" (my/manifolding-atlas-root-dir)))
         (_ (make-directory root t))
         (added (format-time-string "[%Y-%m-%d]"))
         (slug (my/manifolding-atlas-dictionary--slugify word))
         (subdir "")
          (file-name (concat "dictionary/" subdir slug))
         (path-tags (if (string-empty-p subdir)
                        '()
                      (split-string (directory-file-name subdir) "/" t)))
         (manifolding-atlas-create-default-function (lambda (_) '()))
         (prompts (my/manifolding-atlas-collect-prompts-with-fast '(dictionary) '(file)))
         (tags (append (list "dictionary" slug) path-tags
                       (plist-get prompts :tags)))
         (unfoldings (append
                      `(("ADDED" . ,added))
                      (plist-get prompts :unfoldings)))
         (body (plist-get prompts :body))
         (post-apply (plist-get prompts :post-apply))
         (note (my/manifolding-atlas--create-file-wrapper
                word file-name tags unfoldings body)))
    (dolist (fn post-apply)
      (funcall fn note))
    (my/manifolding-atlas-dictionary--open-body-capture note body)
    (my/manifolding-atlas-dictionary-regenerate-dictionary)
    note))
#+end_src

* Dictionary Regeneration
#+begin_src emacs-lisp
(defun my/manifolding-atlas-dictionary--all-words ()
  "Return the sorted, deduplicated list of words from dictionary_words."
  (let ((rows (emacsql (manifolding-atlas-db)
               [:select [word] :from dictionary_words])))
    (sort (delete-dups (mapcar #'car rows)) #'string<)))

(defun my/manifolding-atlas-dictionary-regenerate-dictionary ()
  "Rebuild the flat ispell .pws file from all notes tagged \"dictionary\"."
  (interactive)
  (make-directory (file-name-directory my/manifolding-atlas-dictionary-pws-file) t)
  (let ((words (my/manifolding-atlas-dictionary--all-words)))
    (with-temp-file my/manifolding-atlas-dictionary-pws-file
      (insert (format "personal_ws-1.1 en %d\n" (length words)))
      (dolist (w words)
        (insert w "\n"))))
  (message "Vocab dictionary regenerated (%d words)"
           (length (my/manifolding-atlas-dictionary--all-words))))

;; Point ispell at the generated file.
(with-eval-after-load 'ispell
  (setq ispell-personal-dictionary my/manifolding-atlas-dictionary-pws-file))
#+end_src

(defun my/manifolding-atlas-dictionary--note-files ()
  "Return file paths of all notes tagged \"dictionary\"."
  (mapcar #'manifolding-atlas-note-path
          (manifolding-atlas-db-query-by-tags-some '("dictionary"))))

(defun my/manifolding-atlas-dictionary-drill ()
  "Run an org-drill session scoped to dictionary word files only."
  (interactive)
  (require 'org-drill)
  (let ((files (my/manifolding-atlas-dictionary--note-files)))
    (unless files
      (user-error "No dictionary notes found"))
    (org-drill files)))
