;;; Schema structure

(cl-defstruct (manifolding-atlas-blueprint (:constructor manifolding-atlas-blueprint--create)
                             (:copier nil))
  "A schema describing structured expectations for a class of notes.

Slots:
  name      - symbol identifying the schema
  predicate - function (note) -> non-nil selecting notes it applies to
  fields    - list of field spec plists (see Commentary)
  includes  - names of the schemas this one inherits fields from"
  name
  predicate
  fields
  includes)

;;; Registry

(defvar manifolding-atlas-blueprint--registry (make-hash-table :test 'eq)
  "Registry of defined schemas, keyed by name symbol.")

(defun manifolding-atlas-blueprint--merge-fields (field-lists)
  "Merge FIELD-LISTS, a list of field-spec lists, into a single list.
The lists are processed in order; a later field spec overrides an
earlier one with the same :key, whether the duplicate falls within a
single list or across lists, while an overridden field keeps the
position of its first appearance.  Used to fold the fields of included
schemas into a schema that names them in :include."
  (let ((order nil)
        (by-key (make-hash-table :test 'equal)))
    (dolist (fields field-lists)
      (dolist (field fields)
        (let ((key (plist-get field :key)))
          (unless (gethash key by-key)
            (push key order))
          (puthash key field by-key))))
    (mapcar (lambda (key) (gethash key by-key)) (nreverse order))))

(cl-defun manifolding-atlas-blueprint-define (name &key predicate fields include)
  "Define and register a schema called NAME.

PREDICATE is a function (note) -> non-nil when the schema applies to
the note.  FIELDS is a list of field spec plists (see Commentary for
the supported keys).

INCLUDE names another schema to inherit fields from - a schema name
symbol, a `manifolding-atlas-blueprint' object, or a list of either.  The included
schemas contribute their fields first (multiple includes merged
left-to-right), then FIELDS.  When a :key appears more than once the
later definition wins, so FIELDS overrides an included field and a
later include overrides an earlier one; an overridden field keeps the
position of its first appearance.  Includes are resolved now, so an
included schema must already be registered, and re-defining it later
does not retroactively update schemas that included it.

Registering a schema with an existing NAME replaces it.  Returns the
`manifolding-atlas-blueprint' object."
  (unless (symbolp name)
    (error "Schema name must be a symbol: %S" name))
  (unless (functionp predicate)
    (error "Schema :predicate must be a function"))
  (dolist (field fields)
    (unless (and (plist-member field :key)
                 (stringp (plist-get field :key)))
      (error "Schema field must have a string :key: %S" field)))
  (let* ((includes (cond
                    ((null include) nil)
                    ((or (symbolp include) (manifolding-atlas-blueprint-p include)) (list include))
                    ((listp include) include)
                    (t (error "Schema :include must be a symbol, a schema, or a list of them: %S"
                              include))))
         (inherited (mapcar (lambda (s)
                              (manifolding-atlas-blueprint-fields (manifolding-atlas-blueprint--resolve s)))
                            includes))
         (include-names (mapcar (lambda (s)
                                  (if (manifolding-atlas-blueprint-p s) (manifolding-atlas-blueprint-name s) s))
                                includes))
         (schema (manifolding-atlas-blueprint--create
                  :name name
                  :predicate predicate
                  :includes include-names
                  :fields (manifolding-atlas-blueprint--merge-fields
                           (append inherited (list fields))))))
    (puthash name schema manifolding-atlas-blueprint--registry)
    schema))

(defun manifolding-atlas-blueprint-get (name)
  "Return the registered schema called NAME, or nil."
  (gethash name manifolding-atlas-blueprint--registry))

(defun manifolding-atlas-blueprint-list ()
  "Return the list of registered schema names."
  (let (names)
    (maphash (lambda (k _v) (push k names)) manifolding-atlas-blueprint--registry)
    (nreverse names)))

(defun manifolding-atlas-blueprint-unregister (name)
  "Remove the schema called NAME from the registry."
  (remhash name manifolding-atlas-blueprint--registry))

(defun manifolding-atlas-blueprint--resolve (schema-or-name)
  "Return a `manifolding-atlas-blueprint' from SCHEMA-OR-NAME.

SCHEMA-OR-NAME is either a `manifolding-atlas-blueprint' object or the symbol name of
a registered schema.  Signals an error when a name is not registered."
  (cond
   ((manifolding-atlas-blueprint-p schema-or-name) schema-or-name)
   ((symbolp schema-or-name)
    (or (manifolding-atlas-blueprint-get schema-or-name)
        (error "No schema registered as %S" schema-or-name)))
   (t (error "Not a schema or schema name: %S" schema-or-name))))

;;; Field access

(defun manifolding-atlas-blueprint-field-value (note field)
  "Return the value of FIELD for NOTE, read with the field's declared type.

FIELD is a field spec plist or a bare metadata key string (in which
case the type defaults to `string').  Reads from the note's
already-extracted metadata via `manifolding-atlas-note-meta-get'.  When the field
spec is :multiple, returns a list of values."
  (let* ((spec (if (stringp field) (list :key field) field))
         (key (plist-get spec :key))
         (type (or (plist-get spec :type) 'string)))
    (if (plist-get spec :multiple)
        (manifolding-atlas-note-meta-get-list note key type)
      (manifolding-atlas-note-meta-get note key type))))

;;; Predicate

(defun manifolding-atlas-blueprint-applies-p (note schema-or-name)
  "Return non-nil when SCHEMA-OR-NAME's predicate matches NOTE.

SCHEMA-OR-NAME is a `manifolding-atlas-blueprint' object or a registered schema name."
  (let ((schema (manifolding-atlas-blueprint--resolve schema-or-name)))
    (funcall (manifolding-atlas-blueprint-predicate schema) note)))

;;; Validation

(cl-defstruct (manifolding-atlas-violation (:copier nil))
  "A single schema violation found on a note.

Slots:
  note-id    - id of the offending note
  note-title - title of the offending note
  schema     - name of the schema that was violated
  field      - metadata key of the offending field
  type       - violation type symbol (see `manifolding-atlas-blueprint-validate')
  message    - human-readable description
  value      - the offending value, when applicable"
  note-id note-title schema field type message value)

(defun manifolding-atlas-blueprint--call-or-value (spec note)
  "Resolve SPEC for NOTE: call it when a function, otherwise return it.
Used for field keys like :required and :one-of that accept either a
literal or a function of the note."
  (if (functionp spec) (funcall spec note) spec))

(defun manifolding-atlas-blueprint--coerce (raw type)
  "Coerce the RAW string value to TYPE for comparison."
  (pcase type
    ('number (string-to-number raw))
    ('symbol (intern raw))
    (_ raw)))

(defun manifolding-atlas-blueprint--link-id (raw)
  "Return the note id referenced by RAW, or nil when RAW is not an id link.
Accepts a bracket link \"[[id:UUID][desc]]\" or a bare UUID."
  (cond
   ((string-match org-link-bracket-re raw)
    (let ((link (match-string 1 raw)))
      (when (string-prefix-p "id:" link)
        (string-remove-prefix "id:" link))))
   ((string-match-p (concat "\\`" (or (bound-and-true-p manifolding-atlas-utils--uuid-regexp) "[0-9a-fA-F\\-]\\{36\\}") "\\'") raw)
     raw)))

(defun manifolding-atlas-blueprint--violation (note schema field type message value)
  "Build a `manifolding-atlas-violation' for NOTE on SCHEMA's FIELD of TYPE.
MESSAGE describes the problem and VALUE is the offending value."
  (make-manifolding-atlas-violation
   :note-id (manifolding-atlas-note-id note)
   :note-title (manifolding-atlas-note-title note)
   :schema (manifolding-atlas-blueprint-name schema)
   :field (plist-get field :key)
   :type type
   :message message
   :value value))

(defun manifolding-atlas-blueprint--target-tags-violation (target id field note schema)
  "Return a violation when TARGET is missing FIELD's :target-tags, else nil.

TARGET is the resolved `manifolding-atlas-note' referenced by FIELD on NOTE, and ID
is its id (used in the message and as the offending value).  SCHEMA owns
the field.  All declared tags must be present (all-of); the missing ones
are named in the message."
  (let* ((required (plist-get field :target-tags))
         (missing (and required
                       (cl-remove-if
                        (lambda (tag) (member tag (manifolding-atlas-note-tags target)))
                        required))))
    (when missing
      (manifolding-atlas-blueprint--violation
       note schema field 'invalid-target
       (format "Field %S target %s is missing required tag(s) %S"
               (plist-get field :key) id missing)
       id))))

(defun manifolding-atlas-blueprint--value-violation (raw field note schema)
  "Return a violation for the RAW value of FIELD on NOTE, or nil when valid.
SCHEMA is the owning schema.  Checks, in order: type, reference
resolution and :target-tags (for `note' fields), allowed values, then a
custom :validate function.  Returns the first problem found."
  (let* ((key (plist-get field :key))
         (type (or (plist-get field :type) 'string))
         (one-of (manifolding-atlas-blueprint--call-or-value (plist-get field :one-of) note))
         (validate (plist-get field :validate)))
    (or
     ;; type / reference checks
     (pcase type
       ('number
        (unless (string-match-p "\\`[ \t]*-?[0-9][0-9.]*[ \t]*\\'" raw)
          (manifolding-atlas-blueprint--violation
           note schema field 'wrong-type
           (format "Field %S expected a number, got %S" key raw) raw)))
       ((or 'note 'link)
        (let ((id (manifolding-atlas-blueprint--link-id raw)))
          (cond
           ((null id)
            (manifolding-atlas-blueprint--violation
             note schema field 'wrong-type
             (format "Field %S expected an id link, got %S" key raw) raw))
           ;; `link' fields only check the id-link shape; `note' fields
           ;; additionally resolve the target and enforce :target-tags.
           ((eq type 'note)
            (let ((target (manifolding-atlas-db-get-by-id id)))
              (if (not target)
                  (manifolding-atlas-blueprint--violation
                   note schema field 'invalid-reference
                   (format "Field %S references missing note %s" key id) id)
                (manifolding-atlas-blueprint--target-tags-violation
                 target id field note schema))))))))
     ;; allowed values (not meaningful for note/link references)
     (when (and one-of (not (memq type '(note link))))
       (let ((typed (manifolding-atlas-blueprint--coerce raw type)))
         (unless (member typed one-of)
           (manifolding-atlas-blueprint--violation
            note schema field 'disallowed-value
            (format "Field %S value %S is not one of %S" key typed one-of)
            typed))))
     ;; custom validator
     (when validate
       (let* ((typed (manifolding-atlas-blueprint--coerce raw type))
              (result (funcall validate typed note)))
         (unless (eq result t)
           (manifolding-atlas-blueprint--violation
            note schema field 'invalid-value
            (if (stringp result) result
              (format "Field %S failed validation" key))
            typed)))))))

(defun manifolding-atlas-blueprint--field-violations (note field schema)
  "Return the list of violations for FIELD on NOTE against SCHEMA."
  (let* ((key (plist-get field :key))
         (required (manifolding-atlas-blueprint--call-or-value (plist-get field :required) note))
         (raws (manifolding-atlas-note-meta-get-list note key 'string)))
    (cond
     ((and required (null raws))
      (list (manifolding-atlas-blueprint--violation
             note schema field 'missing-required
             (format "Required field %S is missing" key) nil)))
     ((null raws) nil)
     (t (delq nil
              (mapcar (lambda (raw)
                        (manifolding-atlas-blueprint--value-violation raw field note schema))
                      raws))))))

(defun manifolding-atlas-blueprint-validate (note schema-or-name)
  "Return the list of `manifolding-atlas-violation' for NOTE against SCHEMA-OR-NAME.

SCHEMA-OR-NAME is a `manifolding-atlas-blueprint' object or a registered schema name.
Each field is checked for the following problem types:

- `missing-required'   a required field has no value
- `wrong-type'         a value does not match the field's declared type
- `invalid-reference'  a `note' field links to a non-existent note
- `invalid-target'     a `note' field's target is missing a :target-tags tag
- `disallowed-value'   a value is not in the field's :one-of set
- `invalid-value'      a value is rejected by the field's :validate function

The note is validated as given; the schema's predicate is not consulted
here (see `manifolding-atlas-blueprint-validate-all')."
  (let ((schema (manifolding-atlas-blueprint--resolve schema-or-name)))
    (cl-loop for field in (manifolding-atlas-blueprint-fields schema)
             append (manifolding-atlas-blueprint--field-violations note field schema))))

(defun manifolding-atlas-blueprint-validate-notes (notes schema-or-name)
  "Return all violations for NOTES against SCHEMA-OR-NAME.

NOTES is any list of `manifolding-atlas-note' objects; every note is validated
regardless of the schema's predicate.  To validate only the notes a
schema applies to, filter with `manifolding-atlas-blueprint-applies-p' first, or use
`manifolding-atlas-blueprint-validate-all'."
  (let ((schema (manifolding-atlas-blueprint--resolve schema-or-name)))
    (cl-loop for note in notes
             append (manifolding-atlas-blueprint-validate note schema))))

(defun manifolding-atlas-blueprint-validate-all (schema-or-name)
  "Return all violations for every note matched by SCHEMA-OR-NAME.

Notes are selected with the schema's predicate, then validated.  This
is sugar for `manifolding-atlas-blueprint-validate-notes' over the matching notes."
  (let ((schema (manifolding-atlas-blueprint--resolve schema-or-name)))
    (manifolding-atlas-blueprint-validate-notes
     (manifolding-atlas-db-query (lambda (note) (manifolding-atlas-blueprint-applies-p note schema)))
     schema)))

;;; Authoring

(defun manifolding-atlas-blueprint-applicable (note)
  "Return the names of registered schemas whose predicate matches NOTE.
Schemas are tested in registration order."
  (cl-remove-if-not (lambda (name) (manifolding-atlas-blueprint-applies-p note name))
                    (manifolding-atlas-blueprint-list)))

(defun manifolding-atlas-blueprint-missing-fields (note schema-or-name &optional required-only)
  "Return the fields of SCHEMA-OR-NAME that NOTE has no value for.

Required fields come first (in declared order), then optional ones;
with REQUIRED-ONLY non-nil only required fields are returned.
Requiredness is evaluated against NOTE, so a function :required is
honored.  A field counts as present when NOTE has any value for its
:key, so this never proposes a field the note already carries."
  (let ((schema (manifolding-atlas-blueprint--resolve schema-or-name))
        (required nil)
        (optional nil))
    (dolist (field (manifolding-atlas-blueprint-fields schema))
      (when (null (manifolding-atlas-note-meta-get-list note (plist-get field :key) 'string))
        (if (manifolding-atlas-blueprint--call-or-value (plist-get field :required) note)
            (push field required)
          (unless required-only (push field optional)))))
    (append (nreverse required) (nreverse optional))))

(defun manifolding-atlas-blueprint-note-violations (note)
  "Return all violations of NOTE across the schemas that apply to it.

Validates NOTE against each schema named by `manifolding-atlas-blueprint-applicable'
and concatenates the results.  Returns nil when no schema applies or the
note conforms to all that do.  This is the per-note counterpart to
`manifolding-atlas-blueprint-validate-all' and the shared entry point for surfacing a
single note's violations (sync warnings, in-buffer linting, health
widgets)."
  (cl-loop for name in (manifolding-atlas-blueprint-applicable note)
           append (manifolding-atlas-blueprint-validate note name)))

;;; Collection health

(cl-defstruct (manifolding-atlas-blueprint-health (:constructor manifolding-atlas-blueprint-health--create)
                                    (:copier nil))
  "Aggregated health of a single schema across a set of notes.

Slots:
  schema        - schema name symbol
  covered       - number of notes the schema applies to
  invalid       - number of covered notes with at least one violation
  invalid-notes - list of (note . violations) for the invalid notes
  includes      - schema names this schema inherits fields from
  included-by   - schema names that inherit fields from this schema"
  schema covered invalid invalid-notes includes included-by)

(cl-defun manifolding-atlas-blueprint-collection-health (&optional (notes (manifolding-atlas-db-query)))
  "Return the health of every registered schema across NOTES.

NOTES defaults to all notes in the database.  The result is a list of
`manifolding-atlas-blueprint-health' structures, one per registered schema in
registration order, each carrying how many notes the schema covers, how
many of those are invalid, the invalid notes paired with their
violations, and the schema's include relationships.  Pure apart from the
default `manifolding-atlas-db-query'; pass NOTES to aggregate an explicit set."
  (let ((by-name (make-hash-table :test 'eq))
        (included-by (make-hash-table :test 'eq))
        result)
    (dolist (name (manifolding-atlas-blueprint-list))
      (let ((includes (manifolding-atlas-blueprint-includes (manifolding-atlas-blueprint-get name))))
        (puthash name (manifolding-atlas-blueprint-health--create
                       :schema name :covered 0 :invalid 0
                       :invalid-notes nil :includes includes)
                 by-name)
        (dolist (inc includes)
          (push name (gethash inc included-by)))))
    (dolist (note notes)
      (dolist (name (manifolding-atlas-blueprint-applicable note))
        (let ((h (gethash name by-name)))
          (when h
            (cl-incf (manifolding-atlas-blueprint-health-covered h))
            (let ((vs (manifolding-atlas-blueprint-validate note name)))
              (when vs
                (cl-incf (manifolding-atlas-blueprint-health-invalid h))
                (push (cons note vs) (manifolding-atlas-blueprint-health-invalid-notes h))))))))
    (dolist (name (manifolding-atlas-blueprint-list))
      (let ((h (gethash name by-name)))
        (setf (manifolding-atlas-blueprint-health-invalid-notes h)
              (nreverse (manifolding-atlas-blueprint-health-invalid-notes h)))
        (setf (manifolding-atlas-blueprint-health-included-by h)
              (nreverse (gethash name included-by)))
        (push h result)))
    (nreverse result)))

(require 'manifolding-atlas-note nil t)

(require 'manifolding-atlas-db-extract nil t)

(defcustom manifolding-atlas-db-blueprint-validation-action 'warning
  "How to react when an indexed note violates an applicable schema.

The check runs while a note is written to the database (on save,
external sync, or a full scan), and only when schemas are registered and
one applies to the note - otherwise it does nothing.

- `silent'  index the note, say nothing.
- `warning' index the note, emit a non-blocking warning naming the
            violations (the default).
- `error'   keep the violating note OUT of the database (skip indexing
            it) and emit a warning.  The file on disk is untouched, so
            the note simply stays absent from the database until it is
            fixed and re-synced.

Because files are the source of truth, none of these touch the file;
they only control whether the database mirrors a violating note."
  :type '(choice (const :tag "Index silently" silent)
                 (const :tag "Index and warn" warning)
                 (const :tag "Skip and warn" error))
  :group 'manifolding-atlas)

(defun manifolding-atlas-db-blueprint-validation--format (note violations)
  "Format VIOLATIONS of NOTE into a human-readable warning string."
  (format "Note %S (%s) violates its schema:\n%s"
          (manifolding-atlas-note-title note)
          (manifolding-atlas-note-id note)
          (mapconcat
           (lambda (v)
             (format "  - [%s] %s"
                     (manifolding-atlas-violation-schema v)
                     (manifolding-atlas-violation-message v)))
           violations "\n")))

(defun manifolding-atlas-db-blueprint-validation--warn (note violations level)
  "Emit a warning at LEVEL describing VIOLATIONS of NOTE."
  (lwarn 'manifolding-atlas-blueprint level "%s"
         (manifolding-atlas-db-blueprint-validation--format note violations)))

(defun manifolding-atlas-db-blueprint-validation--filter (note)
  "Validate NOTE for `manifolding-atlas-db-note-index-filter-functions'.

Return non-nil to allow indexing NOTE and nil to skip it, dispatching on
`manifolding-atlas-db-blueprint-validation-action'.  Fast-exits when validation is
`silent' or no schemas are registered, and never affects a note that
conforms or that no schema applies to."
  (if (or (eq manifolding-atlas-db-blueprint-validation-action 'silent)
          (null (manifolding-atlas-blueprint-list)))
      t
    (let ((violations (manifolding-atlas-blueprint-note-violations note)))
      (cond
       ((null violations) t)
       ((eq manifolding-atlas-db-blueprint-validation-action 'error)
        (manifolding-atlas-db-blueprint-validation--warn note violations :error)
        nil)
       (t
        (manifolding-atlas-db-blueprint-validation--warn note violations :warning)
        t)))))

(add-hook 'manifolding-atlas-db-note-index-filter-functions
          #'manifolding-atlas-db-blueprint-validation--filter)

#+begin_src emacs-lisp
(require 'flymake)
(require 'org-element)
(require 'manifolding-atlas-note nil t)
(require 'manifolding-atlas-blueprint nil t)
(require 'manifolding-atlas-buffer nil t)
(require 'manifolding-atlas-db-extract nil t)

(defun manifolding-atlas-blueprint-flymake--note-region (level pos)
  "Return (BEG . END) of the header line of a note at LEVEL and POS.
For a heading-level note (LEVEL above zero) this is the heading line at
POS; for a file-level note it is the =#+title= line, or the first line
when there is none."
  (save-excursion
    (if (and level (> level 0) pos)
        (goto-char pos)
      (goto-char (point-min))
      (unless (re-search-forward "^#\\+title:" nil t)
        (goto-char (point-min))))
    (cons (line-beginning-position) (line-end-position))))

(defun manifolding-atlas-blueprint-flymake--field-region (field bound)
  "Return (BEG . END) of FIELD's metadata line within BOUND, or nil.
BOUND scopes the lookup to a heading subtree (see `manifolding-atlas-buffer-meta')
and is nil for file-level metadata."
  (when-let* ((meta (manifolding-atlas-buffer-meta--get (manifolding-atlas-buffer-meta bound) field))
              (item (car (plist-get meta :items)))
              (beg (org-element-unfolding :begin item)))
    (cons beg (save-excursion (goto-char beg) (line-end-position)))))

(defun manifolding-atlas-blueprint-flymake--violation-region (violation data level)
  "Return (BEG . END) in the current buffer for VIOLATION.
DATA is the extraction plist of the offending note and LEVEL its level.
A value violation points at its field's line; a missing required field
\(which has no line) points at the note's header line."
  (let ((field (manifolding-atlas-violation-field violation))
        (bound (when (and level (> level 0)) (plist-get data :pos))))
    (or (and field
             (not (eq (manifolding-atlas-violation-type violation) 'missing-required))
             (manifolding-atlas-blueprint-flymake--field-region field bound))
        (manifolding-atlas-blueprint-flymake--note-region level (plist-get data :pos)))))

(defun manifolding-atlas-blueprint-flymake--collect ()
  "Return schema-violation diagnostics for the notes in the current buffer.
Each element is (BEG END TYPE TEXT), ready for `flymake-make-diagnostic'."
  (let* ((ast (org-element-parse-buffer))
         (path (or (buffer-file-name) (buffer-name)))
         (file-title (manifolding-atlas-db--extract-file-title ast path))
         (datas (cons (manifolding-atlas-db--extract-file-node ast path (current-buffer) file-title)
                      (manifolding-atlas-db--extract-heading-nodes ast path (current-buffer) file-title)))
         (diagnostics nil))
    (dolist (data datas)
      (when (plist-get data :id)
        (let* ((level (or (plist-get data :level) 0))
               (note (manifolding-atlas-db--note-from-data data path level)))
          (dolist (v (manifolding-atlas-blueprint-note-violations note))
            (let ((region (manifolding-atlas-blueprint-flymake--violation-region v data level)))
              (push (list (car region) (cdr region) :warning
                          (format "[%s] %s"
                                  (manifolding-atlas-violation-schema v)
                                  (manifolding-atlas-violation-message v)))
                    diagnostics))))))
    (nreverse diagnostics)))

(defun manifolding-atlas-blueprint-flymake (report-fn &rest _)
  "Report schema violations in the current buffer to REPORT-FN.
A flymake backend; see `flymake-diagnostic-functions'."
  (funcall report-fn
           (mapcar (lambda (d)
                     (apply #'flymake-make-diagnostic (current-buffer) d))
                   (manifolding-atlas-blueprint-flymake--collect))))

(define-minor-mode manifolding-atlas-blueprint-flymake-mode

(defun my/manifolding-atlas-blueprint--migrate-note-warnings (level pos)
  "Clean WARNING sentinels at one heading anchor; return count deleted."
  (let ((count 0))
    (when (and level (> level 0) pos)
      (goto-char pos)
      (org-back-to-heading t))
    (dolist (unfolding '("STATE_WORK" "STATE_GENERAL" "STATE_INTELLIGENCE"

      (let ((val (org-entry-get nil unfolding)))
        (when (and val (string-match-p "WARNING" val))
          (org-entry-delete nil unfolding)
          (setq count (1+ count)))))

               (equal (org-get-todo-state) "WARNING"))
      (org-todo " ")

      (dolist (info notes)
        (setq count

        (save-buffer))
      (kill-buffer))
    count))
(defun my/manifolding-atlas-blueprint-migrate-warnings ()
  "Delete all unfoldings with value \"WARNING\" throughout the vault.

Under the old prompt system, skipping a prompt stored \"WARNING\" as
the unfolding value to track it in Static Blueprints.  Now schema
validation flags notes that are *missing* the metadata entirely as
`missing-required` violations, so the sentinel is no longer needed.
Deleting the WARNING sentinel lets the description-list metadata
(eg - STATE_WORK :: ACTIVE) become the canonical value."
  (interactive)
  (unless (yes-or-no-p
           "Delete all \"WARNING\" unfolding values throughout the vault? ")
    (user-error "Aborted"))
  (let ((count 0)
        (files (make-hash-table :test #'equal)))
    ;; Group notes by file so we only visit each file once
    (dolist (note (manifolding-atlas-db-query))
      (let ((file (manifolding-atlas-note-path note)))
        (when (and file (file-exists-p file))
          (push (list :level (manifolding-atlas-note-level note)
                      :pos (manifolding-atlas-note-pos note))
                (gethash file files)))))
    (maphash
     (lambda (file notes)
       (setq count
             (+ count (my/manifolding-atlas-blueprint--migrate-file-warnings
                       file notes))))
     files)
     (message "Migration complete: deleted %d WARNING unfolding values (existing metadata preserved)." count)))
