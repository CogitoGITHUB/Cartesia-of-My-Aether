#+title: Manifolding Keyboard — Vundo Map
#+priority: 20

Direct keymap bindings for vundo-mode.

** Vundo
#+begin_src emacs-lisp
(with-eval-after-load 'vundo
  (add-hook 'vundo-mode-hook (lambda () (manifolding-keyboard-set-state nil)))
  (define-key vundo-mode-map (kbd "h") #'vundo-backward)
  (define-key vundo-mode-map (kbd "t") #'vundo-forward)
  (define-key vundo-mode-map (kbd "n") #'vundo-next)
  (define-key vundo-mode-map (kbd "s") #'vundo-previous)
  (define-key vundo-mode-map (kbd "q") #'vundo-quit)
  (define-key vundo-mode-map (kbd "<return>") #'vundo-confirm))
#+end_src
