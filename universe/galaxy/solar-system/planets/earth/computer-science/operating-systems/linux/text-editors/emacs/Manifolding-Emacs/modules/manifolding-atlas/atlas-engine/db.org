#+TITLE: Manifolding Atlas — Database

* Database

The substrate law: org files are the durable truth; =notes= and
=links= are the machine-readable structural substrate — identity
plus relations plus traversal. =mm_nodes=, =mastering_schedule=, and
every plugin table are peer disposable indexes over that substrate,
each rebuilt from files without touching the others. Higher-level
queries compose indexes; no index is authoritative over another, and
the substrate itself stays domain-neutral (it knows graph structure,
never what any subsystem means by it).

(defun my/manifolding-atlas-db-health ()
  "Probe the current backend; return (STATE . HINT).
STATE is CONNECTED, DOWN, AUTH-FAILED, DATABASE-MISSING, or
SCHEMA-OUTDATED. HINT names the exact shell recovery step. Never
signals — infrastructure state must be reportable, not exceptional."
  (condition-case err
      (let ((db (manifolding-atlas-db)))
        (if (null db)
            (cons 'DOWN "pg_ctl -D $PREFIX/var/lib/postgresql start")
          (condition-case nil
              (progn
                (emacsql db [:select [id] :from notes :limit 1])
                (cons 'CONNECTED nil))
            (error (cons 'SCHEMA-OUTDATED "rescan the vault to rebuild indexes")))))
    (error
     (let ((msg (error-message-string err)))
       (cond ((string-match-p "refused\\|could not connect\\|no such host" msg)
              (cons 'DOWN "pg_ctl -D $PREFIX/var/lib/postgresql start"))
             ((string-match-p "auth\\|password" msg)
              (cons 'AUTH-FAILED "check ~/.authinfo entry and pg_hba scram-sha-256"))
             ((string-match-p "does not exist" msg)
              (cons 'DATABASE-MISSING "createdb atlas (after createuser)"))
             (t (cons 'DOWN msg)))))))

(defun manifolding-atlas-db (&optional _args)
  "Return DB connection, opening it if needed."
  (require 'emacsql)
  (if (eq my/manifolding-atlas-db-backend 'pg)
      (unless (and manifolding-atlas-db--connection
                   (ignore-errors (emacsql-live-p manifolding-atlas-db--connection)))
        (setq manifolding-atlas-db--connection
              (my/manifolding-atlas-db--connect-pg)))
    (require 'emacsql-sqlite-builtin)
    (unless (and manifolding-atlas-db--connection
                 (ignore-errors (emacsql-live-p manifolding-atlas-db--connection)))
      (setq manifolding-atlas-db--connection
            (emacsql-sqlite-builtin manifolding-atlas-db-location))))
  (when (and manifolding-atlas-db--connection
             (not (bound-and-true-p my/manifolding-atlas--notes-table-ready)))
    ;; Guarantee `notes' exists for every caller (find/query/create),
    ;; not just update-file path. Direct emacsql on connection avoids
    ;; recursing via ensure-notes-table -> (manifolding-atlas-db).
    (ignore-errors
      (emacsql manifolding-atlas-db--connection
               [:create :table :if :not :exists notes
                ([(id :text :primary-key :not-null)
                  (title :text)
                  (path :text :not-null)
                  (level :integer)
                  (pos :integer)
                  (unfoldings :text)
                  (tags :text)])]))
    (setq my/manifolding-atlas--notes-table-ready t))
  (when (and manifolding-atlas-db--connection
             (not (bound-and-true-p my/manifolding-atlas--links-table-ready)))
    (ignore-errors
      (emacsql manifolding-atlas-db--connection
               [:create :table :if :not :exists links
                ([(source :text :not-null)
                  (dest :text :not-null)]
                 (:primary-key [source dest]))]))
    (setq my/manifolding-atlas--links-table-ready t))
  (when (and manifolding-atlas-db--connection
             (fboundp 'my/wormhole-links-ensure-table)
             (not (bound-and-true-p my/manifolding-atlas--wormhole-links-table-ready)))
    ;; atlas-wormholes.org loads after db.org, so guard with fboundp —
    ;; symmetric with restoration.org's (when (fboundp ...)) call guard.
    (ignore-errors (my/wormhole-links-ensure-table))
    (setq my/manifolding-atlas--wormhole-links-table-ready t))
  manifolding-atlas-db--connection)

(defun manifolding-atlas--register-extractor (extractor)
  "Register EXTRACTOR and keep `manifolding-atlas-db--extractors' sorted by :priority."
  (setq manifolding-atlas-db--extractors
        (sort (cons extractor
                    (cl-remove (plist-get (list :name (manifolding-atlas-extractor-name extractor)) :name)
                               manifolding-atlas-db--extractors
                               :key #'manifolding-atlas-extractor-name :test #'eq))
              (lambda (a b) (< (or (manifolding-atlas-extractor-priority a) 100)
                               (or (manifolding-atlas-extractor-priority b) 100)))))
  ;; de-duplicate by name keep highest priority (last wins, sort stable)
  (setq manifolding-atlas-db--extractors
        (cl-remove-duplicates manifolding-atlas-db--extractors :key #'manifolding-atlas-extractor-name :test #'eq :from-end t)))

(defun my/manifolding-atlas--row-to-note (row)
  "Normalize DB ROW (vector or list) to plist note; tolerate nil/malformed `unfoldings'/`tags' via `ignore-errors' + `read'."
  (if (and (listp row) (plist-get row :id))
      row
    (let ((v (if (vectorp row) (append row nil) row)))
      (list :id (nth 0 v) :title (nth 1 v) :path (nth 2 v)
            :level (or (nth 3 v) 0) :pos (or (nth 4 v) 1)
            :unfoldings (ignore-errors (read (nth 5 v))) :tags (ignore-errors (read (nth 6 v)))
            :aliases nil :primary-title nil :meta nil))))

(cl-defun manifolding-atlas-db--update-file (path)
  "Parse PATH, upsert its `notes' row, and run all registered extractors.
Returns 1 if a note was indexed, 0 otherwise. Handles deletion when file missing."
  (my/manifolding-atlas--ensure-notes-table)
  (if (not (file-exists-p path))
      (progn
        (ignore-errors (emacsql (manifolding-atlas-db) [:delete :from notes :where (= path $s1)] path))
        ;; also purge dependent tables via cascade or explicit
        (dolist (extractor manifolding-atlas-db--extractors)
          (ignore-errors (funcall (manifolding-atlas-extractor-extract-fn extractor)
                                  (make-manifolding-atlas-parse-ctx :path path :ast nil) nil)))
        0)
    ;; Build parse-ctx via existing override if available, else minimal
    (let* ((ctx (if (fboundp 'manifolding-atlas-db--parse-file)
                    (ignore-errors (manifolding-atlas-db--parse-file path))
                  (with-temp-buffer
                    (insert-file-contents path)
                    (let ((ast (ignore-errors (org-element-parse-buffer))))
                      (make-manifolding-atlas-parse-ctx :path path :ast ast)))))
           (ast (and ctx (manifolding-atlas-parse-ctx-ast ctx)))
            (id (with-temp-buffer
                  (insert-file-contents path)
                  (when (re-search-forward "^:ID:[ \t]+\\(.+\\)" nil t)
                    (string-trim (match-string 1)))))
           (title (with-temp-buffer
                    (insert-file-contents path)
                    (when (re-search-forward "^\\*+[ \t]+\\(.*\\)" nil t)
                      (string-trim (match-string 1))))))
      (unless id
        (ignore-errors (emacsql (manifolding-atlas-db) [:delete :from notes :where (= path $s1)] path))
        (cl-return-from manifolding-atlas-db--update-file 0))
      (let ((unfoldings (with-temp-buffer
                     (insert-file-contents path)
                     (when (re-search-forward "^:PROPERTIES:" nil t)
                       (let (alist)
                         (while (re-search-forward "^:\\([^:]+\\):[ \t]*\\(.*\\)" nil t)
                           (let ((k (match-string 1)))
                             (if (string= k "END") (goto-char (point-max))
                               (push (cons k (match-string 2)) alist))))
                         (nreverse alist)))))
            (tags (with-temp-buffer
                    (insert-file-contents path)
                    (when (re-search-forward "^\\*+[ \t]+.*:\\([^:\n]+\\):[ \t]*$" nil t)
                      (split-string (match-string 1) ":" t)))))
        ;; upsert notes row
        (ignore-errors (emacsql (manifolding-atlas-db) [:delete :from notes :where (= id $s1)] id))
        (emacsql (manifolding-atlas-db)
                 [:insert :into notes :values $v1]
                 (vector id (or title (file-name-nondirectory path)) path 0 1
                         (prin1-to-string unfoldings) (prin1-to-string tags)))
        ;; run extractors
        (let ((note-data (list :id id :title title :path path)))
          (dolist (extractor (sort (copy-sequence manifolding-atlas-db--extractors)
                                   (lambda (a b) (< (or (manifolding-atlas-extractor-priority a) 100)
                                                    (or (manifolding-atlas-extractor-priority b) 100)))))
            (let ((fn (manifolding-atlas-extractor-extract-fn extractor)))
              (when fn
                (ignore-errors
                  (let ((result (funcall fn ctx note-data)))
                    (when result (setq note-data result))))))))
        (if (car (emacsql (manifolding-atlas-db) [:select * :from notes :where (= id $s1)] id)) 1 0)))))

*** Orphan prune (state comparison, not hooks)
Shell rm, git checkout, and external tools never fire Emacs hooks,
so deleted files linger as DB ghosts (visible in find completion)
until something diffs. This does: every distinct DB path missing
on disk goes through `manifolding-atlas-db--update-file', which
already deletes the row. File existence is truth — same rule as
the blueprint `.known-keys' reconcile. Admin history untouched.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-db--prune-orphans ()
  "Drop DB note rows for vault files that no longer exist on disk.
Returns pruned count. Never touches files or admin history.
Skips silently when the engine (emacsql/manifolding-atlas-db) is not
loaded yet — boot timers fire before it; the idle repeat catches up."
  (if (not (and (fboundp 'emacsql) (fboundp 'manifolding-atlas-db)
                (fboundp 'manifolding-atlas-db--update-file)))
      0
    (let ((pruned 0))
    (my/manifolding-atlas--ensure-notes-table)
    (let ((paths (ignore-errors
                   (mapcar #'car (emacsql (manifolding-atlas-db)
                                          [:select :distinct [path] :from notes])))))
      (dolist (p paths)
        (when (and (stringp p) (not (file-exists-p p)))
          (ignore-errors (manifolding-atlas-db--update-file p))
          (setq pruned (1+ pruned)))))
    (when (> pruned 0)
      (message "db orphans: pruned %d note(s) for deleted files" pruned))
    pruned)))

(defun manifolding-atlas-db-query (&optional filter)
  "Query notes, filtered by FILTER if given.
Rows whose file no longer exists on disk are dropped before FILTER
is applied, so callers never see stale/deleted entries."
  (let* ((db (ignore-errors (manifolding-atlas-db)))
         (db-result (when db
                      (let ((notes (seq-filter
                                    (lambda (n) (file-exists-p (manifolding-atlas-note-path n)))
                                    (mapcar #'my/manifolding-atlas--row-to-note (emacsql db [:select * :from notes])))))
                        (if filter (seq-filter filter notes) notes)))))
    (if (and db-result (consp db-result))
        db-result
      ;; Fallback: scan vault directly when DB not available or empty (same fallback as elsewhere tonight)
      (let* ((root (my/manifolding-atlas-root-dir))
             (files (ignore-errors (my/manifolding-atlas-db-sync--list-org-files root)))
             notes)
        (dolist (file (or files (directory-files-recursively root "^[^.]" t)))
          (when (and (file-regular-p file)
                     (not (string-match-p "/\\.git/" file))
                     (not (string-match-p "/admin/" file)))
            (let ((id (ignore-errors
                        (with-temp-buffer
                          (insert-file-contents file)
                          (when (re-search-forward "^:ID:[ \t]+\\(.+\\)" nil t)
                            (string-trim (match-string 1)))))))
              (when id
                (let ((title (ignore-errors
                               (with-temp-buffer
                                 (insert-file-contents file)
                                 (when (re-search-forward "^\\*+[ \t]+\\(.*\\)" nil t)
                                   (let ((raw (string-trim (match-string 1))))
                                     (if (string-match "^TODO[ \t]+\\(.*\\)" raw)
                                         (string-trim (match-string 1 raw))
                                       raw)))))))
                  (push (list :id id :title (or title (file-name-nondirectory file))
                              :path file :level 0 :pos 1
                              :tags nil :unfoldings nil :aliases nil :primary-title nil :meta nil)
                        notes))))))
        (if filter
            (seq-filter filter notes)
          notes)))))

** Graph Traversal

Single-hop relations already exist above (links-to, links-from).
This section walks the knowledge graph itself: multi-hop reachability
over =notes= + =links=, domain-neutral and cycle-safe throughout.

The governing distinction, worth protecting: =db-descendants= answers
"What is reachable through knowledge links?" while =mm/subtree-of=
answers "What is contained beneath this node in this map?" Those sets
sometimes look alike but mean different things — semantic
reachability versus positional containment — so the substrate
traverses links and the map traverses placement, and neither knows
about the other's tables. No defun here may reference =mm_nodes=,
stages, order keys, or parent ids; substrate purity is what keeps
later composition powerful (ordered review, orphan triage, relation
tracing, cluster layout — all compose these primitives with domain
indexes rather than living inside them).

*** Breadth-first core
One iterative engine for every multi-hop query: visited set, no
recursion, optional depth cap in edges. Returns visited ids in
discovery order with the starts first, so callers drop the head to
exclude origins.
#+begin_src emacs-lisp
(defun manifolding-atlas-db--bfs (starts step-fn &optional max-depth)
  "Walk from STARTS with STEP-FN (id -> id list), cycle-safe.
Returns visited ids in discovery order, starts first. MAX-DEPTH
caps edges traversed from any start; nil means unbounded."
  (let ((seen (make-hash-table :test 'equal))
        (order nil)
        (frontier (copy-sequence starts))
        (depth 0))
    (dolist (s starts)
      (puthash s t seen)
      (push s order))
    (while (and frontier (or (null max-depth) (< depth max-depth)))
      (let (next)
        (dolist (cur frontier)
          (dolist (nxt (funcall step-fn cur))
            (unless (gethash nxt seen)
              (puthash nxt t seen)
              (push nxt order)
              (push nxt next))))
        (setq frontier (nreverse next)))
      (setq depth (1+ depth)))
    (nreverse order)))
#+end_src

*** Reachability
#+begin_src emacs-lisp
(defun manifolding-atlas-db-descendants (id &optional max-depth)
  "Ids reachable from ID following source→dest links, excluding ID.
Semantic reachability through knowledge links — not map containment
(contrast mm/subtree-of, which walks placement instead)."

(defun manifolding-atlas-db-shortest-path (from to)
  "Id path [FROM ... TO] along links, nil when disconnected.
FROM == TO returns the one-element path (FROM): a thing reaches
itself trivially, which keeps relationship explanations and visual
tracing total instead of special-cased."
  (if (string-equal from to)
      (list from)
    (let ((seen (make-hash-table :test 'equal))
          (prev (make-hash-table :test 'equal))
          (queue (list from))
          found)
      (puthash from t seen)
      (while (and queue (not found))
        (let ((cur (pop queue)))
          (dolist (nxt (manifolding-atlas-db-neighbors cur))
            (unless (gethash nxt seen)
              (puthash nxt t seen)
              (puthash nxt cur prev)
              (if (string-equal nxt to)
                  (setq found t)
                (setq queue (append queue (list nxt))))))))
      (when found
        (let ((path (list to))
              (cur to))
          (while (not (string-equal cur from))
            (setq cur (gethash cur prev))
            (push cur path))
          path)))))

(defun manifolding-atlas-db-link-roots ()
  "Ids that link out but are never linked to: entry points of the graph.
Where a newcomer (or a review walk) would start reading."
  (let ((db (manifolding-atlas-db)))
    (when db
      (let ((dests (make-hash-table :test 'equal))
            (rows (emacsql db [:select [source dest] :from links])))
        (dolist (r rows)
          (puthash (nth 1 r) t dests))
        (seq-uniq (seq-filter (lambda (s) (not (gethash s dests)))
                              (mapcar #'car rows)))))))

(defun manifolding-atlas-db-isolated ()
  "Note ids with no links in either direction: unintegrated cards.
The honest feeder for UNSEEN triage and future \"connect this\"
review prompts."
  (let ((db (manifolding-atlas-db)))
    (when db
      (let ((linked (make-hash-table :test 'equal)))
        (dolist (r (emacsql db [:select [source dest] :from links]))
          (puthash (nth 0 r) t linked)
          (puthash (nth 1 r) t linked))
        (seq-filter (lambda (id) (not (gethash id linked)))
                    (mapcar #'car (emacsql db [:select [id] :from notes])))))))

(defun manifolding-atlas-db-prune-missing ()
  "Remove DB rows whose file no longer exists on disk.

  (interactive)
  (let* ((db (ignore-errors (manifolding-atlas-db)))

(defun manifolding-atlas-select-from (prompt candidates &rest _args)
  "Real select-from: completing-read over CANDIDATES titles, return existing or new plist."
  (require 'seq)
  (let* ((titles (mapcar (lambda (n) (or (manifolding-atlas-note-title n) (plist-get n :title) "")) candidates))
         (choice (completing-read (cond ((string-suffix-p ": " prompt) prompt)
                                         ((string-suffix-p ":" prompt) (concat prompt " "))

(defun manifolding-atlas-db--extract-aliases (unfoldings)
  "Extract aliases from UNFOLDINGS alist.

Looks for unfolding defined by `manifolding-atlas-buffer-alias-unfolding'.
Handles both quoted aliases (with spaces) and unquoted aliases properly.
Strips org links and emphasis markers from each alias.

The unfolding name is upcased before lookup because UNFOLDINGS keys
are stored upcased, so a lowercase or mixed-case
`manifolding-atlas-buffer-alias-unfolding' still matches."
  (when-let* ((aliases-str (cdr (assoc (upcase manifolding-atlas-buffer-alias-unfolding)
                                       unfoldings))))
    (setq aliases-str (string-trim aliases-str))
    (let ((result nil)
          (pos 0))
      (while (< pos (length aliases-str))
        ;; Skip whitespace
        (while (and (< pos (length aliases-str))
                    (= (aref aliases-str pos) ? ))
          (setq pos (1+ pos)))
        (when (< pos (length aliases-str))
          (let ((char (aref aliases-str pos)))
            (cond
             ;; Quoted alias - find matching closing quote
             ((= char ?\")
              (let ((end (string-match "\"" aliases-str (1+ pos))))
                (if end
                    (progn
                      (push (manifolding-atlas-db--strip-emphasis
                             (org-link-display-format
                              (substring aliases-str (1+ pos) end)))
                            result)
                      (setq pos (1+ end)))
                  (error "Unmatched quote in %s" manifolding-atlas-buffer-alias-unfolding))))
             ;; Unquoted alias - find next space or end of string
             (t
              (let ((end (or (string-match " " aliases-str pos)
                             (length aliases-str))))
                (push (manifolding-atlas-db--strip-emphasis
                       (org-link-display-format
                        (substring aliases-str pos end)))
                       result)
                (setq pos end)))))))
      (nreverse result))))
