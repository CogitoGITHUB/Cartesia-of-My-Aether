** Config
#+begin_src emacs-lisp
(setq citar-bibliography '("~/bib/references.bib"))

(setq org-cite-global-bibliography citar-bibliography
      org-cite-insert-processor   'citar
      org-cite-follow-processor   'citar
      org-cite-activate-processor 'citar)

(add-hook 'org-mode-hook
          (lambda () (add-to-list 'completion-at-point-functions #'citar-capf)))
(add-hook 'LaTeX-mode-hook
          (lambda () (add-to-list 'completion-at-point-functions #'citar-capf)))

(with-eval-after-load 'manifolding-atlas
  (defun my/citar-manifolding-atlas-get-candidates (keys)
    "Return alist of (citekey . display) for KEYS that have manifolding-atlas notes."
    (mapcar (lambda (k) (cons k k))
            (seq-filter
             (lambda (k)
               (not (null (manifolding-atlas-db-query-by-meta "citekey" k))))
             keys)))

  (defun my/citar-manifolding-atlas-has-notes (keys)
    "Return hash of citekey -> bool for KEYS."
    (let ((tbl (make-hash-table :test 'equal)))
      (dolist (k keys)
        (puthash k
                 (not (null (manifolding-atlas-db-query-by-meta "citekey" k)))
                 tbl))
      tbl))

  (defun my/citar-manifolding-atlas-open-note (citekey)
    "Open or create a manifolding-atlas note for CITEKEY."
    (let ((note (car (manifolding-atlas-db-query-by-meta "citekey" citekey))))
      (if note
          (manifolding-atlas-visit note)
        (manifolding-atlas-create citekey nil
                       :properties `(("CITEKEY" . ,citekey))))))

  (citar-register-notes-source
   'citar-manifolding-atlas
   (list :name      "Manifolding Atlas Notes"
         :category  'file
         :items     #'my/citar-manifolding-atlas-get-candidates
         :hasitems  #'my/citar-manifolding-atlas-has-notes
         :open      #'my/citar-manifolding-atlas-open-note
         :create    #'my/citar-manifolding-atlas-open-note))

  (setq citar-notes-source 'citar-manifolding-atlas))
#+end_src
