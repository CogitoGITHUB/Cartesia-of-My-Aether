
#+title: Manifolding Atlas — Unfoldings
#+priority: -5

* Unfoldings
:PROPERTIES:
:CATEGORY: editing
:END:

Replaces static property values with Org links to canonical or scoped unfolding notes.
Three unfolding types: ~'closed~ (shared, deduped), ~'scoped~ (per-entity), ~nil~ (plain string).

** Canonicalize Value

#+begin_src emacs-lisp
(defun my/manifolding-atlas--canonicalize (value)
  "Downcase, dasherize, strip non-alphanumeric for use as a filename."
  (let ((s (downcase value)))
    (setq s (replace-regexp-in-string "[^a-z0-9]+" "-" s))
    (replace-regexp-in-string "^-+\\|-+$" "" s)))
#+end_src

  (when (file-exists-p path)

      (insert-file-contents path)

** Tolerant Drawer Access

*** Drawer region locator
Finds the property drawer of the heading at point without org's own
strict parsing — vault notes carry raw wormhole lines in drawers.
#+begin_src emacs-lisp
(defun my/manifolding-atlas--drawer-region ()
  (save-excursion
    (when (condition-case nil (org-back-to-heading t) (error nil))
      (let* ((beg (line-beginning-position))
             (hend (save-excursion
                     (goto-char beg)
                     (forward-line 1)
                     (if (re-search-forward "^\\*+[ \t]" (point-max) t)
                         (line-beginning-position)
                       (point-max)))))
        (goto-char beg)
        (when (re-search-forward "^:PROPERTIES:[ \t]*$" hend t)
          (let ((a (line-beginning-position 2)))
            (when (re-search-forward "^:END:[ \t]*$" hend t)
              (cons a (line-beginning-position)))))))))
#+end_src

*** Drawer read
#+begin_src emacs-lisp
(defun my/manifolding-atlas--drawer-get (key)
  (save-excursion
    (when-let ((r (my/manifolding-atlas--drawer-region)))
      (goto-char (car r))
      (or (when (re-search-forward
                 (format "^:%s:[[:blank:]]+\\(.*\\)$" (regexp-quote key))
                 (cdr r) t)
            (match-string-no-properties 1))
          (progn
            (goto-char (car r))
            (when (re-search-forward
                   "^#\\+wormhole: \\[\\[id:\\([^]]+\\)\\]\\[\\([^]]*\\)\\]\\]"
                   (cdr r) t)
              (let ((label (match-string-no-properties 2)))
                (when (string-equal key (my/manifolding-atlas--key-from-label label))
                  (format "[[id:%s][%s]]"
                          (match-string-no-properties 1) label)))))))))
#+end_src

*** Drawer write
#+begin_src emacs-lisp
(defun my/manifolding-atlas--drawer-put (key value)
  (save-excursion
    (let ((r (my/manifolding-atlas--drawer-region)))
      (if r
          (progn
            (goto-char (car r))
            (if (re-search-forward
                 (format "^:%s:[[:blank:]]*\\(.*\\)$" (regexp-quote key))
                 (cdr r) t)
                (progn (beginning-of-line)
                       (delete-region (line-beginning-position)
                                      (line-end-position))
                       (insert (format ":%s: %s" key value)))
              (goto-char (cdr r))
              (beginning-of-line)
              (insert (format ":%s: %s\n" key value))))
        (org-back-to-heading t)
        (end-of-line)
        (forward-line 1)
        (unless (bolp) (insert "\n"))
        (insert (format ":PROPERTIES:\n:%s: %s\n:END:\n" key value))))))
#+end_src

*** org-entry fallbacks
Drawer storage stays transparent: org-entry-get/put fall through to
the tolerant primitives when the standard paths find nothing.
#+begin_src emacs-lisp
(defun my/manifolding-atlas--org-entry-get-fallback (orig-fn pom property &optional inherit literal-nil)
  (or (funcall orig-fn pom property inherit literal-nil)
      (when (and property
                 (or (null pom)
                     (and (integerp pom)
                          (<= (point-min) pom) (<= pom (point-max)))))
        (save-excursion
          (when (integerp pom) (goto-char pom))
          (my/manifolding-atlas--drawer-get
           (if (symbolp property) (symbol-name property) property))))))

(defun my/manifolding-atlas--org-entry-put-fallback (orig-fn pom property value)
  (if (and (or (null pom)
               (and (integerp pom)
                    (<= (point-min) pom) (<= pom (point-max))))
           (my/manifolding-atlas--drawer-region))
      (my/manifolding-atlas--drawer-put
       (if (symbolp property) (symbol-name property) property)
       value)
    (funcall orig-fn pom property value)))

(advice-add 'org-entry-get :around #'my/manifolding-atlas--org-entry-get-fallback)
(advice-add 'org-entry-put :around #'my/manifolding-atlas--org-entry-put-fallback)
#+end_src

** Ensure Unfolding Note (closed vocab)

#+begin_src emacs-lisp
(defun my/manifolding-atlas--ensure-unfolding-note (property-type value &optional _note)
  "Create or reuse an unfolding note for UNFOLDING-TYPE=VALUE.
When a parent note can be determined (via _NOTE, session, or point),
creates a *scoped* unique file per parent (via `create-scoped-note`);
otherwise falls back to shared vocab file. This makes each unfolding
unique per parent so per-file notes don't leak."
  (when (and value (not (string-equal value "WARNING")))
    (let* ((parent-note (or _note
                            (and (bound-and-true-p my/manifolding-atlas-org-prompt--session)
                                 (plist-get my/manifolding-atlas-org-prompt--session :note))
                            (ignore-errors (my/manifolding-atlas--note-at-point))
                            (ignore-errors
                              (when-let ((m (and (boundp 'my/manifolding-atlas-links--deleted-id)
                                                 my/manifolding-atlas-links--deleted-id))
                                         (note (ignore-errors (manifolding-atlas-db-get-by-id m))))
                                note))))
           (parent-id (and parent-note (ignore-errors (manifolding-atlas-note-id parent-note)))))
      (if parent-id
          ;; Unique per parent — scoped file holds notes for *this* parent only
          (my/manifolding-atlas--create-scoped-note property-type value parent-id _note)
        ;; Fallback shared (legacy / no parent context)
        (let* ((slug (my/manifolding-atlas--canonicalize value))
               (root (expand-file-name (format "admin/unfoldings/%s/" property-type) (my/manifolding-atlas-root-dir)))
               (path (expand-file-name slug root))
               (id (my/manifolding-atlas--note-id-by-path path))
               (key (my/manifolding-atlas--property-key-from-type property-type))
               (tag (replace-regexp-in-string "[^A-Za-z0-9_@]" "_" value))
               (title (format "%s >" key)))
          (if id
              (my/manifolding-atlas--make-id-link id value)
            (make-directory root t)
            (let ((manifolding-atlas-create-default-function (lambda (_) '())))
              (let ((note (manifolding-atlas-create title path
                           :tags (list tag) :unfoldings nil :body nil)))
                (when note
                  (my/manifolding-atlas--make-id-link (manifolding-atlas-note-id note) value))))))))))
#+end_src

(require 'cl-lib)

(defun my/manifolding-atlas--property-put (note key value)
  "Set property KEY to VALUE in NOTE's file on disk using drawer-put."
  (let* ((file (manifolding-atlas-note-path note))
         (level (manifolding-atlas-note-level note))
         (pos   (manifolding-atlas-note-pos note)))

        (save-buffer)))))

*** Heading property scan
Reads plain drawer entries AND in-drawer wormhole lines as
uniform (KEY . VALUE) pairs.
#+begin_src emacs-lisp
(defun my/manifolding-atlas--props-in-region (beg end props)
  "Collect (KEY . VALUE) pairs between BEG/END into PROPS."
  (goto-char beg)
  (while (re-search-forward
          "^:\\([A-Z][A-Z0-9_]*\\):[[:blank:]]+\\(.*\\)$"
          end t)
    (push (cons (match-string-no-properties 1)
                (match-string-no-properties 2))
          props))
  (goto-char beg)
  (while (re-search-forward
          "^#\\+wormhole: \\[\\[id:\\([^]]+\\)\\]\\[\\([^]]*\\)\\]\\]"
          end t)
    (let ((key (my/manifolding-atlas--key-from-label
                (match-string-no-properties 2))))
      (when key
        (push (cons key
                    (format "[[id:%s][%s]]"
                            (match-string-no-properties 1)
                            (match-string-no-properties 2)))
              props))))
  props)

(defun my/manifolding-atlas--heading-properties ()
  (let (props)
    (when (condition-case nil (progn (org-back-to-heading t) t) (error nil))
      (save-excursion
        (let* ((beg (condition-case nil (progn (org-back-to-heading t) (line-beginning-position)) (error (point-min))))
               (end (save-excursion
                      (goto-char beg)
                      (forward-line 1)
                      (if (re-search-forward "^\\*+[ \t]" (point-max) t)
                          (line-beginning-position)
                        (point-max)))))
          (goto-char beg)
          (when (re-search-forward "^:PROPERTIES:" end t)
            (let ((dr-beg (line-beginning-position 2)))
              (when (re-search-forward "^:END:" end t)
                (setq props
                      (my/manifolding-atlas--props-in-region
                       dr-beg (line-beginning-position) props))))))))
    props))
#+end_src

*** Wormhole removal sweep
Drops stale :property wormhole lines of this type, keeping UNFOLDING-ID,
and reports which ids were removed for later cleanup.
#+begin_src emacs-lisp :lexical t
(defun my/manifolding-atlas--remove-other-type-wormholes (property-type unfolding-id)
  "Delete UNFOLDING-TYPE wormhole lines except UNFOLDING-ID; return removed ids."
  (let (removed)
    (goto-char (point-min))
    (while (re-search-forward
            "^#\\+wormhole: \\[\\[id:\\([^]]+\\)\\]\\[" nil t)
      (let ((lid (match-string-no-properties 1)))
        (when (and (my/manifolding-atlas--unfolding-of-type-p property-type lid)
                   (not (string-equal lid unfolding-id)))
          (push lid removed)
          (delete-region (line-beginning-position)
                          (1+ (line-end-position))))))
    removed))
#+end_src

*** Refresh orchestration
#+begin_src emacs-lisp :lexical t
(defun my/manifolding-atlas--refresh-property-wormhole (property-type drawer-value unfolding-id)
  "Replace UNFOLDING-TYPE's current entry IN PLACE -- at whatever
position it already occupies in the drawer, whether a plain value
line or a legacy wormhole -- instead of always appending at
the bottom of the drawer. Writes a `#+wormhole:' line so
`atlas-wormholes-add-all' can render it live; re-picking also
clears legacy wormholes of this type."
  (let (removed insert-marker)
    (save-excursion
      (org-back-to-heading t)
      (let* ((key (my/manifolding-atlas--property-key-from-type property-type))
             (r (my/manifolding-atlas--drawer-region)))
        (when r
          (goto-char (car r))
          (if (re-search-forward
               (format "^:%s:[[:blank:]]*\\(.*\\)$" (regexp-quote key))
               (cdr r) t)
              (progn
                (beginning-of-line)
                (setq insert-marker (copy-marker (point)))
                (delete-region (point) (1+ (line-end-position))))
            (goto-char (car r))
            (let (found)
              (while (and (not found)
                          (re-search-forward
                           "^#\\+wormhole: \\[\\[id:\\([^]]+\\)\\]\\[\\([^]]*\\)\\]\\]"
                           (cdr r) t))
                (let ((label (match-string-no-properties 2)))
                  (when (string-equal key (my/manifolding-atlas--key-from-label label))
                    (setq found t)
                    (beginning-of-line)
                    (setq insert-marker (copy-marker (point)))
                    (delete-region (point) (1+ (line-end-position)))))))))
        (setq removed (my/manifolding-atlas--remove-other-type-wormholes
                       property-type unfolding-id))
        (goto-char (or (and insert-marker (marker-position insert-marker))
                       (my/manifolding-atlas--drawer-insert-point)
                       (point-max)))
        (beginning-of-line)
        (insert (format "#+wormhole: [[id:%s][%s: %s]] :property\n"
                        unfolding-id
                        (string-replace "_" " " key)
                        drawer-value))
        (when insert-marker (set-marker insert-marker nil)))
      (save-buffer))
    removed))
#+end_src

*** Unfolding usage checks and orphan sweep
#+begin_src emacs-lisp :lexical t
(defun my/manifolding-atlas--unfolding-in-use-p (id)
  (let* ((root (my/manifolding-atlas-root-dir))
         (unfoldings-dir (expand-file-name "admin/unfoldings/" root))
         (files (nconc (directory-files-recursively root "\\.org\\'")
                       (directory-files-recursively root "[^./]+\\'")))
         (in-use nil))
    (dolist (f files)
      (unless in-use
        (let ((exp (expand-file-name f)))
          (when (and (not (file-in-directory-p exp unfoldings-dir))
                     (my/manifolding-atlas-links--unfolding-id-in-file-p id exp))
            (setq in-use t)))))
    in-use))

(defun my/manifolding-atlas--cleanup-unused-unfolding (id)
  (when (and id (not (my/manifolding-atlas--unfolding-in-use-p id)))
    (let ((path (ignore-errors (org-id-find-id-file id))))
      (when (and path
                 (file-exists-p path)
                 (file-in-directory-p
                  path
                  (expand-file-name "admin/unfoldings/" (my/manifolding-atlas-root-dir))))
        (message "my/manifolding-atlas: deleting unused unfolding %s" path)
        (delete-file path)))))

(defun my/manifolding-atlas--sweep-orphaned-unfoldings ()
  (interactive)
  (let* ((root (my/manifolding-atlas-root-dir))
         (unfoldings-dir (expand-file-name "admin/unfoldings/" root))
         (files (directory-files-recursively unfoldings-dir "[^./]+\\'" t))
         count)
    (dolist (f files)
      (when (file-regular-p f)
        (with-temp-buffer
          (insert-file-contents f)
          (goto-char (point-min))
          (when (re-search-forward "^:ID:[[:blank:]]+\\(.+\\)" nil t)
            (let ((id (string-trim (match-string-no-properties 1))))
              (unless (my/manifolding-atlas--unfolding-in-use-p id)
                (message "my/manifolding-atlas: deleting unused unfolding %s" f)
                (delete-file f)
                (setq count (1+ (or count 0)))))))))
    (message "my/manifolding-atlas: swept %d orphaned unfolding%s"
             (or count 0) (if (and count (= count 1)) "" "s"))))
#+end_src

*** State post-apply
The state is represented ONLY by an in-drawer #+wormhole: line, so
this drops the plain drawer entry and refreshes that transcription.
Wormhole rendering is stripped first so the drawer edits see raw
#+wormhole: directives; after-save-hook re-renders them.
#+begin_src emacs-lisp :lexical t
(defun my/manifolding-atlas--apply-state-post (property-type drawer-value unfolding-id note)
  "Post-apply wrapper for an unfolding state.
Navigates by NOTE's own path, so it works from any buffer."
  (when-let ((file (manifolding-atlas-note-path note)))
    (with-current-buffer (find-file-noselect file)
      (let ((re-render (bound-and-true-p atlas-wormholes-mode))
            removed)
        (when re-render (atlas-wormholes-remove-all))
        (unwind-protect
            (save-excursion
              (let ((pos (manifolding-atlas-note-pos note)))
                (when (and pos (> pos 0)) (goto-char pos)))
              (setq removed
                    (my/manifolding-atlas--refresh-property-wormhole
                     property-type drawer-value unfolding-id)))
          (when re-render
            (ignore-errors (atlas-wormholes-add-all))))
        (dolist (old-id removed)
          (my/manifolding-atlas--cleanup-unused-unfolding old-id))))))

(defun my/manifolding-atlas--clear-state-transcribes (property-type _note)
  (save-excursion
    (goto-char (point-min))
    (while (re-search-forward
            "^#\\+wormhole: \\[\\[id:\\([^]]+\\)\\]\\[" nil t)
      (let ((lid (match-string-no-properties 1)))
        (when (my/manifolding-atlas--unfolding-of-type-p property-type lid)
          (delete-region (line-beginning-position)
                         (1+ (line-end-position))))))
    (save-buffer)))

 (defun my/manifolding-atlas--make-unfolding-state-to-plist (property-type drawer-key)
  (let ((pt property-type) (dk drawer-key))
    (lambda (val _ctx)

               (unfolding-id (my/manifolding-atlas--id-from-link link)))

                           #'my/manifolding-atlas--apply-state-post
                           pt val unfolding-id))))))))

*** Per-drawer stripping
Swaps each in-drawer wormhole line for a plain `:KEY:` synonym so
org's parser sees ordinary properties; returns a restore cell.
#+begin_src emacs-lisp
(defun my/manifolding-atlas--strip-one-drawer ()
  "Strip wormhole lines in the drawer at point; return restore cell."
  (let* ((pbeg (line-beginning-position 2))
         (pm (save-excursion
               (if (re-search-forward "^:END:[ \t]*$" nil t)
                   (copy-marker (line-beginning-position))
                 nil)))
         restore)
    (when pm
      (goto-char pbeg)
      (while (re-search-forward "^[^:].*$" pm t)
        (let* ((text (match-string 0))
               (m (when (string-match
                         "^#\\+wormhole: \\[\\[id:\\([^]]+\\)\\]\\[\\([^]]*\\)\\]\\]"
                         text)
                    (let* ((label (match-string-no-properties 2 text))
                           (key (my/manifolding-atlas--key-from-label label)))
                      (when key
                        (list key
                              (match-string-no-properties 1 text)
                              label)))))
               (syn (when m
                      (format ":%s: [[id:%s][%s]]"
                              (nth 0 m) (nth 1 m) (nth 2 m)))))
          (push (cons (point-marker) (cons text syn)) restore)
          (delete-region (line-beginning-position)
                         (1+ (line-end-position)))
          (when syn
            (insert syn "\n"))))
      (goto-char pm)
      (set-marker pm nil))
    restore))
#+end_src

*** Restore pass
Puts the raw wormhole lines back exactly where they were.
#+begin_src emacs-lisp
(defun my/manifolding-atlas--restore-stripped-lines (restore)
  (dolist (cell (nreverse restore))
    (goto-char (car cell))
    (when (cddr cell)
      (delete-region (line-beginning-position)
                     (1+ (line-end-position))))
    (insert (cadr cell) "\n")
    (set-marker (car cell) nil)))
#+end_src

** RET on a property -- declarative session, then link
Press RET on a property line inside a :PROPERTIES: drawer. The
registered declarative spec for that key opens as a picking session
(letters jump+mark, RET advances); on finish, the picked value is
linked as an unfolding note and its file opens directly so writing
can start immediately. If the property already holds an unfolding
link, choose to update the linked note directly or change to a
different option. Missing spec is a loud error naming the key --
never a silent empty picker. Everywhere else, RET behaves normally.
#+begin_src emacs-lisp
(defun my/manifolding-atlas--current-property-key ()
  "Return the property KEY on the current line, or nil if not on one."
  (save-excursion
    (beginning-of-line)
    (when (looking-at "^[ \t]*:\\([A-Za-z_]+\\):")
      (match-string-no-properties 1))))



(defun my/manifolding-atlas--open-unfolding-file-capture (unfolding-id)
  "Open UNFOLDING-ID's file using the vault's own capture-style editor,
which already sets :kill-buffer and org-capture-use-indirect-buffer
correctly so C-c C-c restores the original window."
  (let ((file (ignore-errors (org-id-find-id-file unfolding-id))))
    (unless (and file (file-exists-p file))
      (user-error "Could not find the unfolding file for %s" unfolding-id))
    (my/manifolding-atlas--edit-file-in-capture file)))

(defun my/manifolding-atlas--property-pick-live (key property-type)
  "Pick an option for KEY from its declarative spec and go live instantly.
Opens the spec's blueprint file in a capture picker; a letter jump
selects the option, kills the picker (picking saves nothing), then
creates/reuses the unfolding note, writes its id-link as KEY's
value, saves, and opens the unfolding note in a capture for editing.
No options in the file is a loud error -- never a silent empty picker."
  (let* ((spec (gethash property-type my/manifolding-atlas-org-prompt--specs))
         (file (and spec (plist-get spec :file)))
         (marker (point-marker)))
    (unless (and spec (stringp file) (file-exists-p file))
      (user-error "No blueprint file for %s" key))
    (my/manifolding-atlas--edit-file-in-capture file)
    (goto-char (point-min))
    (unless (re-search-forward "^\\*\\* .+$" nil t)
      (org-capture-kill)
      (user-error "No options in %s" (file-name-nondirectory file)))
    (goto-char (point-min))
    (let ((ks-all-windows nil))
      (ks-jump "^\\*\\* .+$"
               :beg (point-min)
               :end (point-max)
               :action
               (lambda (pt)
                 (goto-char pt)
                 (org-back-to-heading t)
                 (let ((final-title (string-trim (org-get-heading t t t t))))
                   (org-capture-kill)
                   (my/manifolding-atlas--property-go-live
                    marker property-type key final-title)))))))

(defun my/manifolding-atlas--property-ensure-context-line (unfolding-id key value)
  "Ensure the unfolding note shows `KEY → VALUE' under its drawer.
Inserts the context line plus blank space as the first body lines,
only when not already present -- re-picks never duplicate it.
Repairs already-glued `:END:KEY → value' into correct spacing, and
ensures exactly one blank line between :END: and context and one
blank after (file ends with trailing newline). Returns non-nil when
inserted or repaired."
  (let ((file (and unfolding-id
                   (let ((f (and (fboundp 'org-id-find-id-file)
                                 (org-id-find-id-file unfolding-id))))
                     (when (and f (file-exists-p f)) f)))))
    (when file
      (with-current-buffer (find-file-noselect file)
        (save-excursion
          (goto-char (point-min))
          (when (re-search-forward "^\\* " nil t)
            (org-back-to-heading t)
            (let ((r (my/manifolding-atlas--drawer-region)))
              (when r
                (let ((desired (format "%s → %s" key value))
                      (desired-re (format "^%s → %s[ \t]*$"
                                          (regexp-quote key)
                                          (regexp-quote value))))
                  ;; Repair glued :END: line if present
                  (goto-char (cdr r))
                  (when (looking-at "^:END:[ \t]*\\(.+\\)$")
                    (let ((trailing (string-trim (match-string 1))))
                      (unless (string-empty-p trailing)
                        (delete-region (match-beginning 1) (line-end-position))
                        ;; If glued trailing was not desired, preserve it as body content
                        (when (and (not (string= trailing desired))
                                   (not (string-empty-p trailing)))
                          (save-excursion
                            (goto-char (cdr r))
                            (forward-line 1)
                            (unless (looking-at "^[ \t]*$")
                              (insert "\n"))
                            (insert trailing "\n")))
                        ;; Ensure END line ends cleanly
                        (goto-char (cdr r))
                        (end-of-line)
                        (unless (bolp) (insert "\n")))))
                  ;; Normalize spacing and ensure context line
                  (goto-char (cdr r))
                  (forward-line 1)
                  (let ((posAfterEnd (point)))
                    ;; If already correctly spaced (blank, desired, blank) do nothing
                    (if (and (looking-at "^[ \t]*$")
                             (save-excursion
                               (forward-line 1)
                               (looking-at desired-re))
                             (save-excursion
                               (forward-line 2)
                               (or (looking-at "^[ \t]*$") (eobp))))
                        nil
                      ;; Remove any existing desired line nearby to avoid duplication
                      (save-excursion
                        (let ((searchEnd (save-excursion (forward-line 5) (point))))
                          (when (re-search-forward desired-re searchEnd t)
                            (delete-region (line-beginning-position)
                                           (min (1+ (line-end-position)) (point-max))))))
                      ;; Remove extra blank lines at insertion point
                      (goto-char posAfterEnd)
                      (while (and (not (eobp)) (looking-at "^[ \t]*$"))
                        (delete-region (point) (min (1+ (line-end-position)) (point-max))))
                      ;; Insert exactly one blank, desired, blank
                      (insert "\n" desired "\n\n")
                      ;; Ensure file ends with trailing newline
                      (goto-char (point-max))
                      (unless (bolp) (insert "\n"))
                      (save-buffer)
                      t)))))))))))

(defun my/manifolding-atlas--property-go-live (marker property-type key final-title)
  "Make FINAL-TITLE live for KEY at MARKER, then open it for editing.
Create/reuse the unfolding note, replace KEY's drawer entry with a
live `#+wormhole:' line via `my/manifolding-atlas--refresh-property-wormhole',
save the origin, and open the unfolding note in a capture."
  (let ((buf (marker-buffer marker)))
    (unless (buffer-live-p buf)
      (user-error "Origin buffer gone"))
    (with-current-buffer buf
      (save-excursion
        (goto-char marker)
        (let* ((parent-note (ignore-errors (my/manifolding-atlas--note-at-point)))
               (link (my/manifolding-atlas--ensure-unfolding-note
                      property-type final-title parent-note))
               (unfolding-id (my/manifolding-atlas--id-from-link link)))
          (unless unfolding-id
            (user-error "Could not create a reference note for %s" final-title))
          (save-window-excursion
            (save-excursion
              (goto-char marker)
              (let ((re-render (bound-and-true-p atlas-wormholes-mode)))
                (when re-render (atlas-wormholes-remove-all))
                (unwind-protect
                    (my/manifolding-atlas--refresh-property-wormhole
                     property-type final-title unfolding-id)
                  (when re-render (atlas-wormholes-add-all)))))
            (save-buffer))
          (dolist (win (get-buffer-window-list buf nil t))
            (unless (window-minibuffer-p win)
              (with-selected-window win
                (font-lock-flush)
                (font-lock-ensure))))
          (setq my/manifolding-atlas--pending-unfolding-refresh
                (cons unfolding-id (buffer-file-name buf)))
          (my/manifolding-atlas--open-unfolding-file-capture unfolding-id))))))

(defvar my/manifolding-atlas--pending-unfolding-refresh nil
  "Cons (UNFOLDING-ID . ORIGIN-FILE) to refresh after an unfolding
capture finalizes. Set when RET opens an unfolding; consumed by
`my/manifolding-atlas--maybe-refresh-unfolding-origin'.")

(defun my/manifolding-atlas--maybe-refresh-unfolding-origin ()
  "After a RET-opened unfolding capture finalizes, refresh its origin.
Syncs the origin link description to the unfolding's current title
and reverts origin windows so edits show immediately. No-op unless
such a capture just finalized. Fully error-isolated."
  (when-let ((p my/manifolding-atlas--pending-unfolding-refresh))
    (setq my/manifolding-atlas--pending-unfolding-refresh nil)
    (let ((uid (car p)) (ofile (cdr p)))
      (condition-case err
          (let* ((uf (and uid (ignore-errors (org-id-find-id-file uid))))
                 (new-title (and uf (ignore-errors
                                      (my/manifolding-atlas-links--note-title-of uf)))))
            (when (and uf new-title (stringp ofile))
              (dolist (link (ignore-errors
                              (my/manifolding-atlas--get-incoming-links-with-descriptions uid)))
                (when (and (string-equal (expand-file-name (plist-get link :source-path))
                                         (expand-file-name ofile))
                           (not (string= (or (plist-get link :description) "") new-title)))
                  (manifolding-atlas--update-link-description
                   ofile (plist-get link :pos) new-title)
                  (with-current-buffer (find-file-noselect ofile)
                    (when (buffer-modified-p) (save-buffer)))
                  (message "Origin link updated to \"%s\"" new-title)))))
        (error (lwarn 'manifolding-atlas :warning "Unfolding origin sync: %s" err)))
      (let ((obuf (and (stringp ofile) (get-file-buffer ofile))))
        (when (buffer-live-p obuf)
          (with-current-buffer obuf
            (save-excursion
              ;; Ensure link description line is visually refreshed without revert
              (font-lock-flush (point-min) (point-max))
              (font-lock-ensure (point-min) (point-max))))
          (dolist (win (get-buffer-window-list obuf nil t))
            (unless (window-minibuffer-p win)
              (with-selected-window win
                (font-lock-flush)
                (font-lock-ensure)))))))))

(add-hook 'org-capture-after-finalize-hook #'my/manifolding-atlas--maybe-refresh-unfolding-origin)

(defun my/manifolding-atlas-property-make-unfolding ()
  "RET on a property: pick from the registered declarative spec, then
link the picked unfolding note and open it directly. If the property
already holds an unfolding link, ask whether to update the linked
note directly or change to a different option."
  (interactive)
  (let ((key (my/manifolding-atlas--current-property-key)))
    (unless key
      (user-error "Not on a property line"))
    (when (member key '("ID"))
      (user-error "Cannot make the ID property an unfolding"))
    (when (zerop (hash-table-count my/manifolding-atlas-org-prompt--specs))
      (my/manifolding-atlas-org-prompt--register-all))
    (let* ((property-type (intern (downcase (string-replace "_" "-" key))))
           (spec (gethash property-type my/manifolding-atlas-org-prompt--specs)))
      (unless spec
        (user-error "No declarative spec registered for %s" key))
      (let* ((raw (or (org-entry-get nil key) ""))
             (existing-link (my/manifolding-atlas--unfolding-link-value raw))
             (existing-id (car existing-link)))
        (if existing-id
            (if (y-or-n-p (format "%s is set -- update the note (y) or change the option (n)? " key))
                (my/manifolding-atlas--open-unfolding-file-capture existing-id)
              (my/manifolding-atlas--property-pick-live key property-type))
          (my/manifolding-atlas--property-pick-live key property-type))))))

(defun my/manifolding-atlas-return-dispatch ()
  "RET dispatcher: make the property at point an unfolding if on one, else
fall back to normal `org-return' behavior everywhere else."
  (interactive)
  (if (my/manifolding-atlas--current-property-key)
      (my/manifolding-atlas-property-make-unfolding)
    (call-interactively #'org-return)))

(with-eval-after-load 'org
  (define-key org-mode-map (kbd "RET") #'my/manifolding-atlas-return-dispatch))
