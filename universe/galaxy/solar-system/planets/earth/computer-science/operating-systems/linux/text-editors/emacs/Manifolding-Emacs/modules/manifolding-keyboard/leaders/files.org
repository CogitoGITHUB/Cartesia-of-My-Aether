  (("f" find-file              "find file"        :exit t)
   ("r" consult-recent-file    "recent files"     :exit t)
   ("p" find-file-other-project "other project"   :exit t)

   ("e" (lambda () (interactive) (find-file (my/last-elisp))) "last elisp file" :exit t))
  "Manage"
  (("R" revert-buffer          "revert"     :exit t)
   ("s" my/save-all            "save all"   :exit t)))
