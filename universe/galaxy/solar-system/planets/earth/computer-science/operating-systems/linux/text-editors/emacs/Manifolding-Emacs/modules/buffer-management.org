* Manual Buffer Cleanup Commands

#+begin_src emacs-lisp
(defun my/kill-orphaned-buffers ()
  "Kill all file-visiting buffers whose file no longer exists on disk."
  (interactive)
  (let ((count 0))
    (dolist (buf (buffer-list))
      (when-let* ((file (buffer-file-name buf)))
        (unless (file-exists-p file)
          (kill-buffer buf)
          (cl-incf count))))
    (message "Killed %d orphaned buffer(s)." count)))

(defun my/kill-all-buffers ()
  "Kill all buffers unconditionally."
  (interactive)
  (dolist (buf (buffer-list))
    (kill-buffer buf))
  (message "All buffers killed."))
#+end_src

* Kill All Buffers on Quit
#+begin_src emacs-lisp

#+end_src
